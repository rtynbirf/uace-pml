   206→  for (const el of all) {
   207→    if (performance.now() > deadline) { out.budgetHit = true; break; }
   208→    if (host && (el === host || host.contains(el))) continue;   // свои узлы исключены
   209→    if (el.childElementCount < 1) continue;
   210→    let cs; try { cs = getComputedStyle(el); } catch (e) { continue; }
   211→    if (cs.display === 'none' || cs.visibility === 'hidden') continue;
   212→    if (cs.position !== 'fixed' && cs.position !== 'sticky') continue;
   213→    const r = el.getBoundingClientRect();
   214→    if (!(r.top < 0.15 * vh && r.height >= 32 && r.height <= 96 && r.width >= 120)) continue;
   215→    out.push({ path: pathOf(el), mode: cs.position, h: Math.round(r.height), top: r.top, x: r.x });
   216→  }
   217→  out.sort((a, b) => a.top - b.top || a.x - b.x);
   218→  return out;
   219→}
   220→function findLandmarks(all) {                   // C3 (позиция B на r2): структурные теги + ландмарк-роли, без role=button
   221→  const out = [];
   222→  const tags = { MAIN: 1, NAV: 1, HEADER: 1, FOOTER: 1, ASIDE: 1 };
   223→  const roles = ['banner', 'main', 'navigation', 'complementary', 'contentinfo', 'search', 'region', 'form'];
   224→  for (const el of all) {
   225→    const role = el.getAttribute('role');
   226→    if (!tags[el.tagName] && roles.indexOf(role) < 0) continue;
   227→    out.push({ tag: el.tagName.toLowerCase(), role: role || null, path: pathOf(el) });
   228→  }
   229→  return out;

   231→function scanSync() {                           // один обход от body (свой хост вне body — самоисключение), бюджет ≤150 мс
   232→  const t0 = performance.now();
   233→  const bodyEl = document.body;
   234→  if (!bodyEl) return { all: [], vh: innerHeight || 800, blockCount: 0, maxDepth: 0, sc: { primary: null, containers: [], truncated: false }, groups: [], mounts: [], landmarks: [], truncatedDom: false, elapsed: 0 };
   235→  const every = bodyEl.querySelectorAll('*');
   236→  const all = [...every].slice(0, CAP.nodes);
   237→  const vh = innerHeight || 800;
   238→  let blockCount = 0, maxDepth = 0;
   239→  (function w(n, d) { if (!n || n.nodeType !== 1) return; blockCount++; maxDepth = Math.max(maxDepth, d); for (let c = n.firstElementChild; c; c = c.nextElementSibling) w(c, d + 1); })(bodyEl, 1);
   240→  const sc = pickPrimary(all, vh);
   241→  const deadline = t0 + RECON_BUDGET_MS;
   242→  const groups = findGroups(all, sc.primary ? sc.primary.path : null, deadline);
   243→  const mounts = findMounts(all, vh, deadline);
   244→  const landmarks = findLandmarks(all);
   245→  const budgetHit = !!(groups.budgetHit || mounts.budgetHit);
   246→  return { all, vh, blockCount, maxDepth, sc, groups, mounts, landmarks, truncatedDom: every.length > CAP.nodes || budgetHit, elapsed: Math.round(performance.now() - t0) };

   248→function scoreGroup(g) {                        // r2 §2: чистая формула «кандидат → вес-факты»
   249→  const u = Math.round(100 * Math.max(0, Math.min(1, (g.uniformity - 0.6) / 0.4)));
   250→  const c = Math.round(100 * Math.min(1, g.count / 40));
   251→  const p = g.inPrimary ? 100 : 0;
   252→  const s = Math.round(100 * Math.max(0, Math.min(1, 1 - g.ratio)));
   253→  const l = g.hasLandmark ? 100 : 0;
   254→  return { u, c, p, s, l, confidence: Math.round(0.35 * u + 0.25 * c + 0.20 * p + 0.10 * s + 0.10 * l) };
   255→}
   256→function resolveRecon(scored) {                 // r2 §3: гейты G1–G4 → found/no-data
   257→  const ok = scored.filter(x => x.inPrimary && x.count >= 6 && x.depth >= 2).sort((a, b) => b.confidence - a.confidence || (a.samplePath < b.samplePath ? -1 : 1));
   258→  if (!ok.length || ok[0].confidence < 75) return { status: 'no-data' };
   259→  const second = ok[1];
   260→  if (second && ok[0].confidence - second.confidence < 15 && second.confidence >= 50) return { status: 'no-data' };
   261→  const t = ok[0];
   262→  return { status: 'found', candidate: { path: t.samplePath, sigHash: t.sigHash, inPrimary: true, confidence: t.confidence, basis: { u: t.u, c: t.c, p: t.p, s: t.s, l: t.l } } };
   263→}
   264→function buildStruct() {                        // конверт r1 (+inPrimary, recon по r2) — форма столбца UPOLY

   269→    g2.ratio = primary ? hx(primary.clientHeight / primary.scrollHeight) : 1;
   270→    try { g2.hasLandmark = !!g.el.closest('main,[role="main"],[role="feed"],[role="log"]'); } catch (e) { g2.hasLandmark = false; }

   273→  const struct = {

   276→    dom: { maxDepth: s.maxDepth, blockCount: s.blockCount, landmarks: s.landmarks.slice(0, CAP.landmarks), groups: raw.map(g => ({ sigHash: g.sigHash, count: g.count, depth: g.depth, uniformity: g.uniformity, samplePath: g.samplePath, inPrimary: g.inPrimary })), truncated: s.truncatedDom || s.groups.length > CAP.groups || s.landmarks.length > CAP.landmarks },

   278→    mut: obsResult ? obsResult.mut : { window: 60, adds: 0, removes: 0, recycles: 0, medianIntervalMs: null, streamCandidate: null },
   279→    anchors: { mountPoints: s.mounts.slice(0, CAP.mountPoints).map(m => ({ path: m.path, mode: m.mode, h: m.h })), persistentPaths: obsResult ? obsResult.persistent : [], stability: obsResult ? obsResult.stability : 0, truncated: s.mounts.length > CAP.mountPoints || (obsResult ? obsResult.persistentTruncated : false) }

   286→function startObservation() {
   287→  if (obsSession) return;
   288→  if (!document.body) { log('err', 'нет document.body — наблюдение отменено'); status('наблюдение: нет body'); return; }
   289→  metrics.obsRuns++;
   290→  const s = { adds: 0, removes: 0, recycles: 0, sigBudget: 500, aborted: false, byPath: new Map(), addIntervals: new Map(), lastAdd: new Map(), removedSigs: new Map(), snapshots: [], navEvents: 0, pausedMs: 0, hiddenAt: 0 };

   293→  setBadge('run');
   294→  status('Наблюдение: осталось 60 с');
   295→  log('info', 'окно наблюдения 60 с запущено (Esc — стоп)');
   296→  s.observer = new MutationObserver(muts => {
   297→    for (const m of muts) {
   298→      const p = m.target && m.target.nodeType === 1 ? prefixPath(m.target, 4) : 'body';
   299→      let rec = s.byPath.get(p);
   300→      if (!rec) { rec = { adds: 0, removes: 0, net: 0 }; s.byPath.set(p, rec); }
   301→      for (const n of m.addedNodes) if (n.nodeType === 1) {
   302→        s.adds++; rec.adds++; rec.net++;
   303→        const t = performance.now(), last = s.lastAdd.get(p);
   304→        if (last != null) { const a = s.addIntervals.get(p) || []; a.push(Math.round(t - last)); s.addIntervals.set(p, a); }
   305→        s.lastAdd.set(p, t);
   306→        if (s.sigBudget > 0) {
   307→          s.sigBudget--;
   308→          const h = sigHash(n), set = s.removedSigs.get(p);
   309→          if (set && set.has(h)) { s.recycles++; set.delete(h); }   // I.4.4: remove→add с равным sigHash
   310→        }
   311→      }
   312→      for (const n of m.removedNodes) if (n.nodeType === 1) {
   313→        s.removes++; rec.removes--; rec.net--;
   314→        if (s.sigBudget > 0) { const set = s.removedSigs.get(p) || s.removedSigs.set(p, new Set()).get(p); if (set.size < 50) set.add(sigHash(n)); }
   315→      }
   316→    }

   319→  const snap = () => { const set = new Set(); (function w(n, d) { if (!n || n.nodeType !== 1 || d > 4 || set.size >= 300) return; if (!host || n !== host) { set.add(pathOf(n)); for (let c = n.firstElementChild; c; c = c.nextElementSibling) w(c, d + 1); } })(document.body, 1); s.snapshots.push(set); };

   322→  s.navHandler = () => { s.navEvents++; snap(); };
   323→  s.op = history.pushState; s.orp = history.replaceState;
   324→  history.pushState = function () { s.navHandler(); return s.op.apply(this, arguments); };
   325→  history.replaceState = function () { s.navHandler(); return s.orp.apply(this, arguments); };
   326→  addEventListener('popstate', s.navHandler);
   327→  s.visHandler = () => { if (document.hidden) s.hiddenAt = Date.now(); else if (s.hiddenAt) { s.pausedMs += Date.now() - s.hiddenAt; s.hiddenAt = 0; } };
   328→  document.addEventListener('visibilitychange', s.visHandler);
   329→  s.tick = setInterval(() => {
   330→    const left = Math.max(0, Math.ceil((s.endsAt + s.pausedMs - Date.now()) / 1000));
   331→    status('Наблюдение: осталось ' + left + ' с');
   332→    if (left <= 0) finishObservation(false);
   333→  }, 1000);

   335→function finishObservation(aborted, silent) {
   336→  const s = obsSession;
   337→  if (!s) return;
   338→  obsSession = null;
   339→  clearInterval(s.tick); clearInterval(s.snapTimer);
   340→  s.observer.disconnect();
   341→  document.removeEventListener('visibilitychange', s.visHandler);
   342→  removeEventListener('popstate', s.navHandler);
   343→  history.pushState = s.op; history.replaceState = s.orp;
   344→  const top = [...s.byPath.entries()].sort((a, b) => b[1].adds - a[1].adds)[0];
   345→  const med = top ? median(s.addIntervals.get(top[0]) || []) : null;
   346→  const stream = top ? (top[1].adds >= 3 && top[1].net > 0 && med !== null && med >= 50 && med <= 5000) : false; // I.4.3
   347→  obsResult = {
   348→    mut: { window: 60, adds: s.adds, removes: s.removes, recycles: s.recycles, medianIntervalMs: med, streamCandidate: stream },
   349→    persistent: [], stability: 0, persistentTruncated: false
   350→  };
   351→  if (s.snapshots.length) {                      // I.5.2: присутствуют в ≥90% снапшотов
   352→    const total = s.snapshots.length, cnt = new Map();
   353→    for (const set of s.snapshots) for (const p of set) cnt.set(p, (cnt.get(p) || 0) + 1);
   354→    const per = [...cnt.entries()].filter(x => x[1] / total >= 0.9).sort((a, b) => b[1] - a[1] || (a[0] < b[0] ? -1 : 1));
   355→    obsResult.persistent = per.slice(0, CAP.persistentPaths).map(x => x[0]);
   356→    obsResult.persistentTruncated = per.length > CAP.persistentPaths;
   357→    obsResult.stability = hx(obsResult.persistent.length / Math.max(1, Math.min(s.snapshots[0].size, 20)));
   358→  }
   359→  if (!silent) {                                 // silent: pagehide — авто-разведка при выгрузке глушится (мелочь B-6)
   360→    if (aborted) log('warn', 'наблюдение прервано (Esc): частичный отчёт принят');
   361→    else log('ok', 'наблюдение завершено: adds ' + s.adds + ', recycles ' + s.recycles + ', поток ' + (stream ? 'да' : 'нет'));
   362→    status(aborted ? 'Наблюдение прервано' : 'Наблюдение завершено');
   363→    Promise.resolve().then(runRecon).catch(e => log('err', 'авто-разведка: ' + (e && e.message))); // E1-гигиена: рендер терпит обе формы scroll
   364→  }
   365→}
