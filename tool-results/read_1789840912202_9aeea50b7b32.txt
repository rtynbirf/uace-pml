   102→function rgbToOklch(rgb) {
   103→  const [r, g, b] = rgb.map(srgbLin);
   104→  const l = 0.4122214708 * r + 0.5363325363 * g + 0.0514459929 * b;
   105→  const m = 0.2119034982 * r + 0.6806995451 * g + 0.1073969566 * b;
   106→  const s = 0.0883024619 * r + 0.2817188376 * g + 0.6299787005 * b;
   107→  const L = 0.2104542553 * Math.cbrt(l) + 0.7936177850 * Math.cbrt(m) - 0.0040720468 * Math.cbrt(s);
   108→  const A = 1.9779984951 * Math.cbrt(l) - 2.4285922050 * Math.cbrt(m) + 0.4505937099 * Math.cbrt(s);
   109→  const B = 0.0259040371 * Math.cbrt(l) + 0.7827717662 * Math.cbrt(m) - 0.8086757660 * Math.cbrt(s);
   110→  const C = Math.sqrt(A * A + B * B);
   111→  return { L, C, H: (Math.atan2(B, A) * 180 / Math.PI + 360) % 360 };

   113→function oklchToRgb(L, C, H) {

   119→          f(-1.2684380046 * l + 2.6097574011 * m - 0.3413193965 * s),

   131→  const sys = rgbToOklch(sysRgb);
   132→  if (sys.C < 0.04 || sys.L < 0.05 || sys.L > 0.98) return null;   // сатурационный страж: серый/чёрный/белый
   133→  let best = null;
   134→  for (let L = 0.15; L <= 0.90; L += 0.01) {     // проход 1: фон↔акцент ≥4.5; проход 2: текст-на-акценте ≥4.5
   135→    const rgb = oklchToRgb(L, Math.min(sys.C, 0.16), sys.H);
   136→    const r1 = contrast(bgRgb, rgb);
   137→    if (r1 < 4.5) continue;
   138→    const txWhite = contrast(rgb, [1, 1, 1]), txBlack = contrast(rgb, [0.055, 0.06, 0.075]);
   139→    const tx = txWhite >= txBlack ? ['#ffffff', txWhite] : ['#0e0f12', txBlack];

   142→    if (!best || score > best.score) best = { hex: rgbToHex(rgb), txHex: tx[0], ratio: round2(r1), score };
   143→  }
   144→  return best;

   441→function findGroups(all, primaryPath, deadline) {         // I-1: рекуррентные группы (I.2.2–I.2.5) + inPrimary (r2 3-a)
   442→  const out = [];
   443→  for (const parent of all) {
   444→    if (performance.now() > deadline) { out.budgetHit = true; break; }   // I.8.1: бюджет ≤150 мс
   445→    if (parent.childElementCount < 3) continue;
   446→    const kids = [...parent.children];
   447→    const bySig = new Map();
   448→    for (const k of kids) {
   449→      const h = sigHash(k);
   450→      if (!bySig.has(h)) bySig.set(h, []);
   451→      bySig.get(h).push(k);
   452→    }
   453→    let best = null;
   454→    for (const [h, arr] of bySig) if (!best || arr.length > best.arr.length) best = { h, arr }; // модальная группа
   455→    if (!best || best.arr.length < 3) continue;
   456→    const u = best.arr.length / kids.length;
   457→    if (u < 0.6) continue;
   458→    let pr;
   459→    try { pr = parent.getBoundingClientRect(); } catch (e) { continue; }
   460→    let sumH = 0, hMin = 1e9, hMax = 0, ok = true;
   461→    for (const k of best.arr) {
   462→      const r = k.getBoundingClientRect();
   463→      if (r.height < 32) { ok = false; break; }   // I.2.3: каждый ребёнок ≥32px
   464→      sumH += r.height;
   465→      hMin = Math.min(hMin, r.height); hMax = Math.max(hMax, r.height);
   466→    }
   467→    if (!ok || sumH < 0.4 * pr.height) continue;  // отсечение toolbar/иконочных рядов
   468→    out.push({ sigHash: best.h, count: best.arr.length, depth: subtreeDepth(best.arr[0], 6), uniformity: round2(u), samplePath: pathOf(best.arr[0]), parentPath: pathOf(parent), inPrimary: !!primaryPath && (pathOf(best.arr[0]) === primaryPath || pathOf(best.arr[0]).startsWith(primaryPath + '>')), el: best.arr[0] });
   469→  }
   470→  out.sort((a, b) => b.count * b.uniformity - a.count * a.uniformity || (a.parentPath < b.parentPath ? -1 : 1)); // I.2.5
   471→  return out;
   472→}

   514→function scoreGroup(g) {                        // r2 §2: чистая формула «кандидат → вес-факты»
   515→  const u = Math.round(100 * clamp((g.uniformity - 0.6) / 0.4, 0, 1));

   517→  const p = g.inPrimary ? 100 : 0;
   518→  const s = Math.round(100 * clamp(1 - g.ratio, 0, 1));
   519→  const l = g.hasLandmark ? 100 : 0;
   520→  return { u, c, p, s, l, confidence: Math.round(0.35 * u + 0.25 * c + 0.20 * p + 0.10 * s + 0.10 * l) };
   521→}

   552→/* ---- I-3 + I-4b: пассивное окно 60 с; Esc — стоп; агрегация в idle-режиме браузера ---- */
   553→function startObservation() {
   554→  if (obsSession) return;
   555→  if (!document.body) { log('err', 'нет document.body — наблюдение отменено'); setStatus('наблюдение: нет body'); return; }
   556→  metrics.obsRuns++;
   557→  const s = { adds: 0, removes: 0, recycles: 0, sigBudget: 500, aborted: false, byPath: new Map(), addIntervals: new Map(), lastAdd: new Map(), removedSigs: new Map(), snapshots: [], navEvents: 0, pausedMs: 0, hiddenAt: 0 };
   558→  s.endsAt = Date.now() + OBS_WINDOW_MS;
   559→  obsSession = s;
   560→  setBadge('run'); syncBusy();
   561→  setStatus('Наблюдение: осталось 60 с');
   562→  log('info', 'окно наблюдения 60 с запущено (Esc — стоп)');
   563→  s.observer = new MutationObserver(muts => {
   564→    for (const m of muts) {
   565→      const p = m.target && m.target.nodeType === 1 ? prefixPath(m.target, 4) : 'body';
   566→      let rec = s.byPath.get(p);
   567→      if (!rec) { rec = { adds: 0, removes: 0, net: 0 }; s.byPath.set(p, rec); }
   568→      for (const n of m.addedNodes) if (n.nodeType === 1) {
   569→        s.adds++; rec.adds++; rec.net++;
   570→        const t = performance.now(), last = s.lastAdd.get(p);
   571→        if (last != null) { const a = s.addIntervals.get(p) || []; a.push(Math.round(t - last)); s.addIntervals.set(p, a); }
   572→        s.lastAdd.set(p, t);
   573→        if (s.sigBudget > 0) {
   574→          s.sigBudget--;
   575→          const h = sigHash(n), set = s.removedSigs.get(p);
   576→          if (set && set.has(h)) { s.recycles++; set.delete(h); }   // I.4.4: remove→add с равным sigHash
   577→        }
   578→      }
   579→      for (const n of m.removedNodes) if (n.nodeType === 1) {
   580→        s.removes++; rec.removes--; rec.net--;
   581→        if (s.sigBudget > 0) { const set = s.removedSigs.get(p) || s.removedSigs.set(p, new Set()).get(p); if (set.size < 50) set.add(sigHash(n)); }
   582→      }
   583→    }
   584→  });
   585→  s.observer.observe(document.body, { childList: true, subtree: true, characterData: false, attributes: false });
   586→  const snap = () => { const set = new Set(); (function w(n, d) { if (!n || n.nodeType !== 1 || d > 4 || set.size >= 300) return; if (!host || n !== host) { set.add(pathOf(n)); for (let c = n.firstElementChild; c; c = c.nextElementSibling) w(c, d + 1); } })(document.body, 1); s.snapshots.push(set); };
   587→  s.snap = snap; snap();                         // I.5.2: снапшот 0 + каждые 10 с, глубина ≤4, cap 300
   588→  s.snapTimer = setInterval(() => { if (!document.hidden && s.snapshots.length < 30) snap(); }, 10000);
   589→  s.navHandler = () => { s.navEvents++; snap(); };
   590→  s.op = history.pushState; s.orp = history.replaceState;
   591→  history.pushState = function () { s.navHandler(); return s.op.apply(this, arguments); };
   592→  history.replaceState = function () { s.navHandler(); return s.orp.apply(this, arguments); };
   593→  addEventListener('popstate', s.navHandler);
   594→  s.visHandler = () => { if (document.hidden) s.hiddenAt = Date.now(); else if (s.hiddenAt) { s.pausedMs += Date.now() - s.hiddenAt; s.hiddenAt = 0; } };
   595→  document.addEventListener('visibilitychange', s.visHandler);
   596→  s.tick = setInterval(() => {
   597→    const left = Math.max(0, Math.ceil((s.endsAt + s.pausedMs - Date.now()) / 1000));
   598→    setStatus(`Наблюдение: осталось ${left} с`);
   599→    if (left <= 0) finishObservation(false);
   600→  }, 1000);
   601→}
   602→function finishObservation(aborted) {
   603→  const s = obsSession;
   604→  if (!s) return;
   605→  obsSession = null;
   606→  clearInterval(s.tick); clearInterval(s.snapTimer);
   607→  s.observer.disconnect();
   608→  document.removeEventListener('visibilitychange', s.visHandler);
   609→  removeEventListener('popstate', s.navHandler);
   610→  history.pushState = s.op; history.replaceState = s.orp;
   611→  const top = [...s.byPath.entries()].sort((a, b) => b[1].adds - a[1].adds)[0];
   612→  const med = top ? median(s.addIntervals.get(top[0]) || []) : null;
   613→  const stream = top ? (top[1].adds >= 3 && top[1].net > 0 && med !== null && med >= 50 && med <= 5000) : false; // I.4.3
   614→  obsResult = {
   615→    mut: { window: 60, adds: s.adds, removes: s.removes, recycles: s.recycles, medianIntervalMs: med, streamCandidate: stream },
   616→    persistent: [], stability: 0, persistentTruncated: false
   617→  };
   618→  if (s.snapshots.length) {                      // I.5.2: присутствуют в ≥90% снапшотов
   619→    const total = s.snapshots.length, cnt = new Map();
   620→    for (const set of s.snapshots) for (const p of set) cnt.set(p, (cnt.get(p) || 0) + 1);
   621→    const per = [...cnt.entries()].filter(x => x[1] / total >= 0.9).sort((a, b) => b[1] - a[1] || (a[0] < b[0] ? -1 : 1));
   622→    obsResult.persistent = per.slice(0, CAP.persistentPaths).map(x => x[0]);
   623→    obsResult.persistentTruncated = per.length > CAP.persistentPaths;
   624→    obsResult.stability = round2(obsResult.persistent.length / Math.max(1, Math.min(s.snapshots[0].size, 20)));
   625→  }
   626→  if (aborted) log('warn', 'наблюдение прервано (Esc): частичный отчёт принят');
   627→  else log('ok', `наблюдение завершено: adds ${s.adds}, recycles ${s.recycles}, поток ${stream ? 'да' : 'нет'}`);
   628→  setStatus(aborted ? 'Наблюдение прервано' : 'Наблюдение завершено');
   629→  withBusy(() => runRecon());
   630→}
