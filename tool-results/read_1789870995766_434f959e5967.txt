   617→   617→function collectShadowRoots(root) {
   618→   618→  const roots = []; const seen = new Set();
   619→   619→  const nowMs = Date.now();
   620→   620→  const cached = (!root && _srCache && (nowMs - _srCacheAt) < 1500) ? _srCache : null;
   621→   621→  if (cached) return cached.slice();
   622→   622→  /* v28.2: полностью итеративный обход (стек) — рекурсия устранена (риск stack overflow);
   623→   623→     закрытые (closed) shadowRoot недоступны и/или mode!=='open' — пропускаем осознанно */
   624→   624→  const stack = [root || document];
   625→   625→  seen.add(root || document);
   626→   626→  let visited = 0;
   627→   627→  while (stack.length) {
   628→   628→    const cur = stack.pop();
   629→   629→    if (!cur || (cur.nodeType !== 9 && cur.nodeType !== 1 && cur.nodeType !== 11)) continue;
   630→   630→    let all; try { all = cur.querySelectorAll('*'); } catch (e) { continue; }

   632→   632→      if (++visited > SR_NODE_CAP) { stack.length = 0; break; }
   633→   633→      const el = all[i];
   634→   634→      let sr = null;
   635→   635→      try { sr = el.shadowRoot; } catch (e) {}
   636→   636→      if (!sr || seen.has(sr)) continue;
   637→   637→      if (sr.mode && sr.mode !== 'open') { seen.add(sr); continue; }
   638→   638→      seen.add(sr); roots.push(sr); stack.push(sr);
   639→   639→    }
   640→   640→  }
