   139→    for (let n = 2; n <= 4; n++) {
   140→      if (s.length % n !== 0) continue;
   141→      const chunk = s.slice(0, s.length / n);
   142→      if (!chunk) continue;
   143→      let rep = true;
   144→      for (let i = 1; i < n; i++) { if (s.slice(i * chunk.length, (i + 1) * chunk.length) !== chunk) { rep = false; break; } }

   679→    function tableToMarkdown(table) {
   680→      let rows;
   681→      try { rows = Array.from(table.querySelectorAll('tr')); } catch (e) { return ''; }
   682→      if (!rows.length) return '';
   683→      // v26: поддержка colspan — объединённые ячейки дополняются пустыми, колонки не уезжают
   684→      const matrix = rows.map((row) => {
   685→        const cells = [];
   686→        try {
   687→          Array.from(row.querySelectorAll('th,td')).forEach((cell) => {
   688→            const t = clean(cell.innerText || cell.textContent || '').replace(/\|/g, '\\|').replace(/\n/g, ' ');
   689→            const cs = Math.max(1, parseInt(cell.getAttribute('colspan') || '1', 10) || 1);
   690→            cells.push(t);
   691→            for (let k = 1; k < cs; k++) cells.push('');
   692→          });
   693→        } catch (e) {}
   694→        return cells;
   695→      });
   696→      const width = Math.max(1, ...matrix.map((r) => r.length));
   697→      const norm = matrix.map((r) => { const c = r.slice(); while (c.length < width) c.push(''); return c; });
   698→      const head = norm[0];
   699→      let out = '\n\n| ' + head.join(' | ') + ' |\n| ' + head.map(() => '---').join(' | ') + ' |\n';
   700→      out += norm.slice(1).map((r) => '| ' + r.join(' | ') + ' |').join('\n') + '\n\n';
   701→      return out;
   702→    }
   703→    return clean(walk(root));

   873→  async function waitStable(prevFp, prevHref) {
   874→    const start = Date.now();
   875→    let lastFp = '', lastChangeTime = Date.now();
   876→    let hrefChanged = !!(prevHref && location.href !== prevHref);
   877→    while (Date.now() - start < CFG.MAX_WAIT) {
   878→      if (CANCEL) return false;
   879→      await sleep(CFG.POLL);
   880→      if (!hrefChanged && prevHref && location.href !== prevHref) { hrefChanged = true; lastChangeTime = Date.now(); }
   881→      const cur = fingerprint();
   882→      if (cur && cur !== prevFp) {
   883→        if (cur !== lastFp) { lastFp = cur; lastChangeTime = Date.now(); }
   884→        else if (Date.now() - lastChangeTime >= CFG.STABLE_MS) return true;
   885→      } else if (hrefChanged && Date.now() - lastChangeTime >= CFG.STABLE_MS) {
   886→        return true;
   887→      }
   888→      if (!hrefChanged && !lastFp && Date.now() - start > 6000) return false;
   889→    }
   890→    const fp = fingerprint();
   891→    return (!!fp && fp !== prevFp) || hrefChanged;
   892→  }

  1056→  const CRC_TABLE = (function () {
  1057→    const t = new Uint32Array(256);
  1058→    for (let n = 0; n < 256; n++) {
  1059→      let c = n;
  1060→      for (let k = 0; k < 8; k++) c = (c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1);
  1061→      t[n] = c >>> 0;
  1062→    }
  1063→    return t;
  1064→  })();
  1065→  function crc32(u8) {
  1066→    let c = 0xFFFFFFFF;
  1067→    for (let i = 0; i < u8.length; i++) c = CRC_TABLE[(c ^ u8[i]) & 0xFF] ^ (c >>> 8);
  1068→    return (c ^ 0xFFFFFFFF) >>> 0;
  1069→  }
  1070→  function dosDateTime(d) {
  1071→    const time = ((d.getHours() << 11) | (d.getMinutes() << 5) | Math.floor(d.getSeconds() / 2)) & 0xFFFF;
  1072→    const date = (((d.getFullYear() - 1980) & 0x7f) << 9) | ((d.getMonth() + 1) << 5) | d.getDate();
  1073→    return { time: time, date: date };
  1074→  }
  1075→  // entries: [{name, data:string|Uint8Array, date?}] -> Blob (application/zip)
  1076→  function zipStore(entries) {
  1077→    const enc = new TextEncoder();
  1078→    const localChunks = [], centralChunks = [];
  1079→    let offset = 0;
  1080→    const d = new Date();
  1081→    const { time, date } = dosDateTime(d);
  1082→    for (const e of entries) {
  1083→      const name = enc.encode(e.name);
  1084→      const data = (e.data instanceof Uint8Array) ? e.data : enc.encode(String(e.data));
  1085→      const crc = crc32(data);
  1086→      const lh = new DataView(new ArrayBuffer(30));
  1087→      lh.setUint32(0, 0x04034b50, true);
  1088→      lh.setUint16(4, 20, true);
  1089→      lh.setUint16(6, 0x0800, true);
  1090→      lh.setUint16(8, 0, true);
  1091→      lh.setUint16(10, time, true);
  1092→      lh.setUint16(12, date, true);
  1093→      lh.setUint32(14, crc, true);
  1094→      lh.setUint32(18, data.length, true);
  1095→      lh.setUint32(22, data.length, true);
  1096→      lh.setUint16(26, name.length, true);
  1097→      lh.setUint16(28, 0, true);
  1098→      localChunks.push(new Uint8Array(lh.buffer), name, data);
  1099→      const ch = new DataView(new ArrayBuffer(46));
  1100→      ch.setUint32(0, 0x02014b50, true);
  1101→      ch.setUint16(4, 20, true);
  1102→      ch.setUint16(6, 20, true);
  1103→      ch.setUint16(8, 0x0800, true);
  1104→      ch.setUint16(10, 0, true);
  1105→      ch.setUint16(12, time, true);
  1106→      ch.setUint16(14, date, true);
  1107→      ch.setUint32(16, crc, true);
  1108→      ch.setUint32(20, data.length, true);
  1109→      ch.setUint32(24, data.length, true);
  1110→      ch.setUint16(28, name.length, true);
  1111→      ch.setUint16(30, 0, true);
  1112→      ch.setUint16(32, 0, true);
  1113→      ch.setUint16(34, 0, true);
  1114→      ch.setUint16(36, 0, true);
  1115→      ch.setUint32(38, 0, true);
  1116→      ch.setUint32(42, offset, true);
  1117→      centralChunks.push(new Uint8Array(ch.buffer), name);
  1118→      offset += 30 + name.length + data.length;
  1119→    }
  1120→    let cdSize = 0;
  1121→    for (const c of centralChunks) cdSize += c.length;
  1122→    const eocd = new DataView(new ArrayBuffer(22));
  1123→    eocd.setUint32(0, 0x06054b50, true);
  1124→    eocd.setUint16(4, 0, true);
  1125→    eocd.setUint16(6, 0, true);
  1126→    eocd.setUint16(8, entries.length, true);
  1127→    eocd.setUint16(10, entries.length, true);
  1128→    eocd.setUint32(12, cdSize, true);
  1129→    eocd.setUint32(16, offset, true);
  1130→    eocd.setUint16(20, 0, true);
  1131→    return new Blob(localChunks.concat(centralChunks, [new Uint8Array(eocd.buffer)]), { type: 'application/zip' });
  1132→  }
