   162→   162→  const segs = pathSegs(el).slice(1);
   163→   163→  let p = 'body' + (segs.length ? '>' + segs.join('>') : '');
   164→   164→  if (p.length > 120) p = '~' + p.slice(-119); // усечение слева, хвост глубже важнее
   165→   165→  return p;
   166→   166→};
   167→   167→const prefixPath = (el, d) => pathSegs(el).slice(0, d).join('>');
   168→   168→const sigOf = el => {                          // r2 §5: pre-order теги ВЕРХНИМ регистром через '>', глубина ≤3
   169→   169→  const out = [];
   170→   170→  (function walk(n, d) {
   171→   171→    if (!n || n.nodeType !== 1 || d > 3) return;
   172→   172→    out.push(n.tagName.toUpperCase());
   173→   173→    for (let c = n.firstElementChild; c; c = c.nextElementSibling) walk(c, d + 1);
   174→   174→  })(el, 1);
   175→   175→  return out.join('>');
   176→   176→};
   177→   177→const sigHash = el => hex8(FNV1a32(sigHashIn(sigOf(el))));
   178→   178→const subtreeDepth = (el, cap) => {            // фактическая глубина поддерева (cap 6)
   179→   179→  let max = 1;
   180→   180→  (function walk(n, cur) {
   181→   181→    if (cur > cap) return;
   182→   182→    for (let c = n.firstElementChild; c; c = c.nextElementSibling) { max = Math.max(max, cur + 1); walk(c, cur + 1); }
   183→   183→  })(el, 1);
   184→   184→  return max;
   185→   185→};
   186→   186→const median = arr => {                        // медиана целых; null если пусто
   187→   187→  const a = arr.filter(Number.isFinite).sort((x, y) => x - y);
   188→   188→  if (!a.length) return null;
   189→   189→  const m = a.length >> 1;
   190→   190→  return a.length % 2 ? a[m] : Math.round((a[m - 1] + a[m]) / 2);
   191→   191→};

   193→   193→function pickPrimary(all, vh) {                // I-2: каскад primary (I.3.2) + контейнеры — столбец UPOLY
   194→   194→  const cands = [];
   195→   195→  for (const el of all) {
   196→   196→    if (el.scrollHeight <= el.clientHeight + 4) continue;
   197→   197→    let ov = 'visible';
   198→   198→    try { ov = getComputedStyle(el).overflowY; } catch (e) {}
   199→   199→    if (ov !== 'auto' && ov !== 'scroll' && ov !== 'overlay') continue;

   201→   201→    try { main = !!el.closest('main,[role="main"]'); } catch (e) {}
   202→   202→    cands.push({ el, path: pathOf(el), clientHeight: el.clientHeight, scrollHeight: el.scrollHeight, visibleShare: el.clientHeight / vh, insideMain: main });
   203→   203→  }
   204→   204→  cands.sort((a, b) => b.visibleShare - a.visibleShare || b.scrollHeight - a.scrollHeight || b.clientHeight - a.clientHeight || (b.insideMain - a.insideMain) || (a.path < b.path ? -1 : 1));
   205→   205→  return { primary: cands[0] || null, containers: cands.slice(0, CAP.containers).map(c => ({ path: c.path, scrollHeight: c.scrollHeight, clientHeight: c.clientHeight })), truncated: cands.length > CAP.containers };

   207→   207→function findGroups(all, primaryPath, deadline) {   // I-1: модальная группа на родителя — столбец UPOLY (+ inPrimary r2 3-a)
   208→   208→  const out = [];
   209→   209→  for (const parent of all) {
   210→   210→    if (performance.now() > deadline) { out.budgetHit = true; break; }
   211→   211→    if (parent.childElementCount < 3) continue;
   212→   212→    const kids = [...parent.children];
   213→   213→    const bySig = new Map();
   214→   214→    for (const k of kids) { const h = sigHash(k); if (!bySig.has(h)) bySig.set(h, []); bySig.get(h).push(k); }
   215→   215→    let best = null;
   216→   216→    for (const [h, arr] of bySig) if (!best || arr.length > best.arr.length) best = { h, arr };
   217→   217→    if (!best || best.arr.length < 3) continue;
   218→   218→    const u = best.arr.length / kids.length;
   219→   219→    if (u < 0.6) continue;
   220→   220→    let pr;
   221→   221→    try { pr = parent.getBoundingClientRect(); } catch (e) { continue; }
   222→   222→    let sumH = 0, ok = true;
   223→   223→    for (const k of best.arr) {
   224→   224→      const r = k.getBoundingClientRect();
   225→   225→      if (r.height < 32) { ok = false; break; }   // I.2.3: каждый ребёнок ≥32px
   226→   226→      sumH += r.height;
   227→   227→    }
   228→   228→    if (!ok || sumH < 0.4 * pr.height) continue;  // отсечение toolbar/иконочных рядов
   229→   229→    out.push({ sigHash: best.h, count: best.arr.length, depth: subtreeDepth(best.arr[0], 6), uniformity: hx(u), samplePath: pathOf(best.arr[0]), parentPath: pathOf(parent), inPrimary: !!primaryPath && (pathOf(best.arr[0]) === primaryPath || pathOf(best.arr[0]).startsWith(primaryPath + '>')), el: best.arr[0] });
   230→   230→  }
   231→   231→  out.sort((a, b) => b.count * b.uniformity - a.count * a.uniformity || (a.parentPath < b.parentPath ? -1 : 1)); // I.2.5
   232→   232→  return out;
   233→   233→}
   234→   234→function findMounts(all, vh, deadline) {            // I-4a: кандидаты монтирования — столбец UPOLY + свои узлы вне
   235→   235→  const out = [];

   237→   237→    if (performance.now() > deadline) { out.budgetHit = true; break; }

   246→   246→  }
   247→   247→  out.sort((a, b) => a.top - b.top || a.x - b.x);

   258→   258→  }

   261→   261→function scanSync() {                           // один обход от body (свой хост вне body — самоисключение), бюджет ≤150 мс

   278→   278→function scoreGroup(g) {                        // r2 §2: чистая формула «кандидат → вес-факты»
   279→   279→  const u = Math.round(100 * Math.max(0, Math.min(1, (g.uniformity - 0.6) / 0.4)));
   280→   280→  const c = Math.round(100 * Math.min(1, g.count / 40));
   281→   281→  const p = g.inPrimary ? 100 : 0;
   282→   282→  const s = Math.round(100 * Math.max(0, Math.min(1, 1 - g.ratio)));
   283→   283→  const l = g.hasLandmark ? 100 : 0;
   284→   284→  return { u, c, p, s, l, confidence: Math.round(0.35 * u + 0.25 * c + 0.20 * p + 0.10 * s + 0.10 * l) };
   285→   285→}
   286→   286→function resolveRecon(scored) {                 // r2 §3: гейты G1–G4 → found/no-data
   287→   287→  const ok = scored.filter(x => x.inPrimary && x.count >= 6 && x.depth >= 2).sort((a, b) => b.confidence - a.confidence || (a.samplePath < b.samplePath ? -1 : 1));
   288→   288→  if (!ok.length || ok[0].confidence < 75) return { status: 'no-data' };
   289→   289→  const second = ok[1];
   290→   290→  if (second && ok[0].confidence - second.confidence < 15 && second.confidence >= 50) return { status: 'no-data' };
   291→   291→  const t = ok[0];
   292→   292→  return { status: 'found', candidate: { path: t.samplePath, sigHash: t.sigHash, inPrimary: true, confidence: t.confidence, basis: { u: t.u, c: t.c, p: t.p, s: t.s, l: t.l } } };
   293→   293→}
   294→   294→function buildStruct() {                        // конверт r1 (+inPrimary, recon по r2) — форма столбца UPOLY

   337→   337→const downsample = (st, budget) => { // D2Snap (arXiv 08.2025): coverage + selection под бюджет; детерминированно

   350→   350→const compareStructs = (a, b) => { // Broder/minhash-линейка (1997): точный Jaccard по sigHash групп (наборы ≤12)
