   182→const pathSegs = el => { // сегменты от body (без body/html) — формат столбца UPOLY
   183→  const segs = []; let n = el;
   184→  while (n && n.nodeType === 1 && n.tagName !== 'BODY' && n.tagName !== 'HTML') {
   185→    let i = 1, s = n;
   186→    while ((s = s.previousElementSibling)) if (s.tagName === n.tagName) i++;
   187→    segs.unshift(n.tagName.toLowerCase() + (i > 1 ? ':nth(' + i + ')' : ''));
   188→    n = n.parentElement;
   189→  }
   190→  return ['body'].concat(segs);
   191→};
   192→const pathOf = el => {                         // структурный путь ≤120 симв. (r1 §0): nth(i) только при i>1
   193→  const segs = pathSegs(el).slice(1);
   194→  let p = 'body' + (segs.length ? '>' + segs.join('>') : '');
   195→  if (p.length > 120) p = '~' + p.slice(-119); // усечение слева, хвост глубже важнее
   196→  return p;
   197→};
   198→const prefixPath = (el, d) => pathSegs(el).slice(0, d).join('>');
   199→const sigOf = el => {                          // r2 §5: pre-order теги ВЕРХНИМ регистром через '>', глубина ≤3
   200→  const out = [];
   201→  (function walk(n, d) {
   202→    if (!n || n.nodeType !== 1 || d > 3) return;
   203→    out.push(n.tagName.toUpperCase());
   204→    for (let c = n.firstElementChild; c; c = c.nextElementSibling) walk(c, d + 1);
   205→  })(el, 1);
   206→  return out.join('>');
   207→};
   208→const sigHash = el => hex8(FNV1a32(sigHashIn(sigOf(el))));
   209→const subtreeDepth = (el, cap) => {            // фактическая глубина поддерева (cap 6)
   210→  let max = 1;
   211→  (function walk(n, cur) {
   212→    if (cur > cap) return;
   213→    for (let c = n.firstElementChild; c; c = c.nextElementSibling) { max = Math.max(max, cur + 1); walk(c, cur + 1); }
   214→  })(el, 1);
   215→  return max;
   216→};
   217→const median = arr => {                        // медиана целых; null если пусто
   218→  const a = arr.filter(Number.isFinite).sort((x, y) => x - y);
   219→  if (!a.length) return null;
   220→  const m = a.length >> 1;
   221→  return a.length % 2 ? a[m] : Math.round((a[m - 1] + a[m]) / 2);
   222→};

   224→function pickPrimary(all, vh) {                // I-2: каскад primary (I.3.2) + контейнеры — столбец UPOLY
   225→  const cands = [];

   227→    if (el.scrollHeight <= el.clientHeight + 4) continue;
   228→    let ov = 'visible';
   229→    try { ov = getComputedStyle(el).overflowY; } catch (e) {}
   230→    if (ov !== 'auto' && ov !== 'scroll' && ov !== 'overlay') continue;

   232→    try { main = !!el.closest('main,[role="main"]'); } catch (e) {}
   233→    cands.push({ el, path: pathOf(el), clientHeight: el.clientHeight, scrollHeight: el.scrollHeight, visibleShare: el.clientHeight / vh, insideMain: main });
   234→  }
   235→  cands.sort((a, b) => b.visibleShare - a.visibleShare || b.scrollHeight - a.scrollHeight || b.clientHeight - a.clientHeight || (b.insideMain - a.insideMain) || (a.path < b.path ? -1 : 1));
   236→  return { primary: cands[0] || null, containers: cands.slice(0, CAP.containers).map(c => ({ path: c.path, scrollHeight: c.scrollHeight, clientHeight: c.clientHeight })), truncated: cands.length > CAP.containers };
   237→}
   238→function findGroups(all, primaryPath, deadline) {   // I-1: модальная группа на родителя — столбец UPOLY (+ inPrimary r2 3-a)
   239→  const out = [];
   240→  for (const parent of all) {
   241→    if (performance.now() > deadline) { out.budgetHit = true; break; }
   242→    if (parent.childElementCount < 3) continue;
   243→    const kids = [...parent.children];
   244→    const bySig = new Map();
   245→    for (const k of kids) { const h = sigHash(k); if (!bySig.has(h)) bySig.set(h, []); bySig.get(h).push(k); }
   246→    let best = null;
   247→    for (const [h, arr] of bySig) if (!best || arr.length > best.arr.length) best = { h, arr };
   248→    if (!best || best.arr.length < 3) continue;
   249→    const u = best.arr.length / kids.length;
   250→    if (u < 0.6) continue;
   251→    let pr;
   252→    try { pr = parent.getBoundingClientRect(); } catch (e) { continue; }
   253→    let sumH = 0, ok = true;
   254→    for (const k of best.arr) {
   255→      const r = k.getBoundingClientRect();
   256→      if (r.height < 32) { ok = false; break; }   // I.2.3: каждый ребёнок ≥32px
   257→      sumH += r.height;
   258→    }
   259→    if (!ok || sumH < 0.4 * pr.height) continue;  // отсечение toolbar/иконочных рядов
   260→    out.push({ sigHash: best.h, count: best.arr.length, depth: subtreeDepth(best.arr[0], 6), uniformity: hx(u), samplePath: pathOf(best.arr[0]), parentPath: pathOf(parent), inPrimary: !!primaryPath && (pathOf(best.arr[0]) === primaryPath || pathOf(best.arr[0]).startsWith(primaryPath + '>')), el: best.arr[0] });
   261→  }
   262→  out.sort((a, b) => b.count * b.uniformity - a.count * a.uniformity || (a.parentPath < b.parentPath ? -1 : 1)); // I.2.5
   263→  return out;
   264→}
   265→function findMounts(all, vh, deadline) {            // I-4a: кандидаты монтирования — столбец UPOLY + свои узлы вне
   266→  const out = [];
   267→  for (const el of all) {
   268→    if (performance.now() > deadline) { out.budgetHit = true; break; }
   269→    if (host && (el === host || host.contains(el))) continue;   // свои узлы исключены
   270→    if (el.childElementCount < 1) continue;
   271→    let cs; try { cs = getComputedStyle(el); } catch (e) { continue; }
   272→    if (cs.display === 'none' || cs.visibility === 'hidden') continue;
   273→    if (cs.position !== 'fixed' && cs.position !== 'sticky') continue;
   274→    const r = el.getBoundingClientRect();
   275→    if (!(r.top < 0.15 * vh && r.height >= 32 && r.height <= 96 && r.width >= 120)) continue;
   276→    out.push({ path: pathOf(el), mode: cs.position, h: Math.round(r.height), top: r.top, x: r.x });
   277→  }
   278→  out.sort((a, b) => a.top - b.top || a.x - b.x);
   279→  return out;
   280→}
   281→function findLandmarks(all) {                   // C3 (позиция B на r2): структурные теги + ландмарк-роли, без role=button
   282→  const out = [];

   284→  const roles = ['banner', 'main', 'navigation', 'complementary', 'contentinfo', 'search', 'region', 'form'];
   285→  for (const el of all) {
   286→    const role = el.getAttribute('role');
   287→    if (!tags[el.tagName] && roles.indexOf(role) < 0) continue;
   288→    out.push({ tag: el.tagName.toLowerCase(), role: role || null, path: pathOf(el) });
   289→  }
   290→  return out;

   292→function scanSync() {                           // один обход от body (свой хост вне body — самоисключение), бюджет ≤150 мс
   293→  const t0 = performance.now();
   294→  const bodyEl = document.body;
   295→  if (!bodyEl) return { all: [], vh: innerHeight || 800, blockCount: 0, maxDepth: 0, sc: { primary: null, containers: [], truncated: false }, groups: [], mounts: [], landmarks: [], truncatedDom: false, elapsed: 0 };
   296→  const every = bodyEl.querySelectorAll('*');
   297→  const all = [...every].slice(0, CAP.nodes);
   298→  const vh = innerHeight || 800;
   299→  let blockCount = 0, maxDepth = 0;
   300→  (function w(n, d) { if (!n || n.nodeType !== 1) return; blockCount++; maxDepth = Math.max(maxDepth, d); for (let c = n.firstElementChild; c; c = c.nextElementSibling) w(c, d + 1); })(bodyEl, 1);
   301→  const sc = pickPrimary(all, vh);
   302→  const deadline = t0 + RECON_BUDGET_MS;
   303→  const groups = findGroups(all, sc.primary ? sc.primary.path : null, deadline);
   304→  const mounts = findMounts(all, vh, deadline);
   305→  const landmarks = findLandmarks(all);
   306→  const budgetHit = !!(groups.budgetHit || mounts.budgetHit);
   307→  return { all, vh, blockCount, maxDepth, sc, groups, mounts, landmarks, truncatedDom: every.length > CAP.nodes || budgetHit, elapsed: Math.round(performance.now() - t0) };
   308→}
   309→function scoreGroup(g) {                        // r2 §2: чистая формула «кандидат → вес-факты»
   310→  const u = Math.round(100 * Math.max(0, Math.min(1, (g.uniformity - 0.6) / 0.4)));
   311→  const c = Math.round(100 * Math.min(1, g.count / 40));
   312→  const p = g.inPrimary ? 100 : 0;
   313→  const s = Math.round(100 * Math.max(0, Math.min(1, 1 - g.ratio)));
   314→  const l = g.hasLandmark ? 100 : 0;
   315→  return { u, c, p, s, l, confidence: Math.round(0.35 * u + 0.25 * c + 0.20 * p + 0.10 * s + 0.10 * l) };
   316→}
   317→function resolveRecon(scored) {                 // r2 §3: гейты G1–G4 → found/no-data
   318→  const ok = scored.filter(x => x.inPrimary && x.count >= 6 && x.depth >= 2).sort((a, b) => b.confidence - a.confidence || (a.samplePath < b.samplePath ? -1 : 1));
   319→  if (!ok.length || ok[0].confidence < 75) return { status: 'no-data' };
   320→  const second = ok[1];
   321→  if (second && ok[0].confidence - second.confidence < 15 && second.confidence >= 50) return { status: 'no-data' };
   322→  const t = ok[0];
   323→  return { status: 'found', candidate: { path: t.samplePath, sigHash: t.sigHash, inPrimary: true, confidence: t.confidence, basis: { u: t.u, c: t.c, p: t.p, s: t.s, l: t.l } } };
   324→}

   346→/* ===================== УСВОЕНИЕ ЛАНДШАФТА 27.5.0 (аддитивно; вне хэш-конверта r2 §5) =====================

   368→const downsample = (st, budget) => { // D2Snap (arXiv 08.2025): coverage + selection под бюджет; детерминированно
   369→  const B = Math.max(60, budget == null ? 400 : budget | 0), items = [];
   370→  const push = (prio, line) => items.push({ prio, line, cost: Math.ceil(line.length / 4) }); // «токен» ≈ 4 символа
   371→  push(0, 'page ' + ((st.meta && st.meta.urlHash) || '?'));
   372→  ((st.dom && st.dom.landmarks) || []).forEach((l, i) => push(1, (l.role || l.tag) + ' [ref=l' + (i + 1) + '] ' + l.path));
   373→  ((st.anchors && st.anchors.mountPoints) || []).forEach((m, i) => push(2, 'mount ' + m.mode + ' h' + m.h + ' ' + m.path));
   374→  ((st.dom && st.dom.groups) || []).forEach((g, i) => push(3, 'group ' + g.sigHash + ' x' + g.count + ' d' + g.depth + (g.inPrimary ? ' *' : '') + ' ' + g.samplePath));
   375→  items.sort((a, b) => a.prio - b.prio); // стабильная сортировка: внутри приоритета — канонный порядок struct
   376→  const kept = []; let used = 0, dropped = 0;
   377→  for (const it of items) if (used + it.cost <= B) { kept.push(it.line); used += it.cost; } else dropped++;
   378→  if (dropped) kept.push('… +' + dropped + ' узлов за бюджетом ' + B);

   381→const compareStructs = (a, b) => { // Broder/minhash-линейка (1997): точный Jaccard по sigHash групп (наборы ≤12)
   382→  const ga = new Set(((((a || {}).dom || {}).groups) || []).map(g => g.sigHash));
   383→  const gb = new Set(((((b || {}).dom || {}).groups) || []).map(g => g.sigHash));
   384→  let inter = 0; for (const h of ga) if (gb.has(h)) inter++;
   385→  const union = ga.size + gb.size - inter;
   386→  return { jaccard: union ? hx(inter / union) : 1, sim: simBetween(a, b), inter, groupsA: ga.size, groupsB: gb.size };

   388→/* ---- I-3 + I-4b: пассивное окно 60 с; Esc — стоп; снапшот 0 + каждые 10 с (столбец UPOLY) ---- */
   389→function startObservation() {
   390→  if (obsSession) return;
   391→  if (!document.body) { log('err', 'нет document.body — наблюдение отменено'); status('наблюдение: нет body'); return; }
   392→  metrics.obsRuns++;
   393→  const s = { adds: 0, removes: 0, recycles: 0, sigBudget: 500, aborted: false, byPath: new Map(), addIntervals: new Map(), lastAdd: new Map(), removedSigs: new Map(), snapshots: [], navEvents: 0, pausedMs: 0, hiddenAt: 0 };
   394→  s.endsAt = Date.now() + OBS_WINDOW_MS;
   395→  obsSession = s;
   396→  setBadge('run');
   397→  status('Наблюдение: осталось 60 с');
   398→  log('info', 'окно наблюдения 60 с запущено (Esc — стоп)');
   399→  s.observer = new MutationObserver(muts => {
   400→    for (const m of muts) {
   401→      const p = m.target && m.target.nodeType === 1 ? prefixPath(m.target, 4) : 'body';
   402→      let rec = s.byPath.get(p);
   403→      if (!rec) { rec = { adds: 0, removes: 0, net: 0 }; s.byPath.set(p, rec); }
   404→      for (const n of m.addedNodes) if (n.nodeType === 1) {
   405→        s.adds++; rec.adds++; rec.net++;
   406→        const t = performance.now(), last = s.lastAdd.get(p);
   407→        if (last != null) { const a = s.addIntervals.get(p) || []; a.push(Math.round(t - last)); s.addIntervals.set(p, a); }
   408→        s.lastAdd.set(p, t);
   409→        if (s.sigBudget > 0) {
   410→          s.sigBudget--;
   411→          const h = sigHash(n), set = s.removedSigs.get(p);
   412→          if (set && set.has(h)) { s.recycles++; set.delete(h); }   // I.4.4: remove→add с равным sigHash
   413→        }
   414→      }
