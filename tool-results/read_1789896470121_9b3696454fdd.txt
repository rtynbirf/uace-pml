   239→/* --- L0.4 dedupeNested O(n×depth) (§7) — семантика v28.3.0: контейнер
   240→   (держит ≥2 кандидатов ближайшим уровнем) удаляется, его предметы остаются;
   241→   вложенность в не-контейнер-кандидата — дубликат, удаляется. Вместо тройных
   242→   проходов O(n³) — ближайший-кандидат-предок за O(глубину). --- */
   243→function dedupeNested(items, getParent) {
   244→  const gp = getParent || (el => el.parentElement);
   245→  const arr = Array.from(items).slice(0, 400);
   246→  if (arr.length < 2) return arr;
   247→  const set = new Set(arr);
   248→  const nearAnc = el => { let p = gp(el); while (p) { if (set.has(p)) return p; p = gp(p); } return null; };
   249→  const heldBy = new Map();
   250→  for (const el of arr) { const a = nearAnc(el); if (a) heldBy.set(a, (heldBy.get(a) || 0) + 1); }
   251→  const containers = new Set();
   252→  for (const [a, c] of heldBy) if (c >= 2) containers.add(a);
   253→  const out = [];
   254→  for (const el of arr) {
   255→    if (containers.has(el)) continue;            /* контейнер-«зонтик» долой */
   256→    const a = nearAnc(el);
   257→    if (a && !containers.has(a)) continue;       /* вложен в кандидата-дубликата */
   258→    out.push(el);                                 /* предмет удалённого контейнера */
   259→  }
   260→  return out;
   261→}

   487→const simhash64 = feats => {
   488→  const a1 = new Array(32).fill(0), a2 = new Array(32).fill(0);
   489→  for (const f of feats) {
   490→    const h1 = fnv1a32('SIM1>' + f.k), h2 = fnv1a32('SIM2>' + f.k), w = Math.max(1, Math.min(9, f.w | 0));
   491→    for (let i = 0; i < 32; i++) { if ((h1 >>> i) & 1) a1[i] += w; else a1[i] -= w; if ((h2 >>> i) & 1) a2[i] += w; else a2[i] -= w; }
   492→  }
   493→  let v1 = 0, v2 = 0;
   494→  for (let i = 0; i < 32; i++) { if (a1[i] > 0) v1 |= 1 << i; if (a2[i] > 0) v2 |= 1 << i; }
   495→  return hx(v1) + hx(v2);

   497→const hammingHex = (a, b) => {
   498→  if (typeof a !== 'string' || typeof b !== 'string' || a.length !== 16 || b.length !== 16) return -1;

   573→async function collectShadowRoots(root) {
   574→  const roots = [], seen = new Set();
   575→  let nodes = 0, truncated = false;
   576→  const start = Date.now();
   577→  const topRoot = root || document.documentElement || document;
   578→  seen.add(topRoot);
   579→  const stack = [topRoot];
   580→  while (stack.length) {
   581→    if (roots.length >= CFG.SR_ROOT_CAP || nodes >= CFG.SR_NODE_CAP || Date.now() - start > CFG.SR_TIME_MS) { truncated = true; break; }
   582→    const cur = stack.pop();
   583→    let walker = null;
   584→    try { walker = document.createTreeWalker(cur, NodeFilter.SHOW_ELEMENT); } catch (e) { journal.debug('runtime', 'treewalker-fail', String(e && e.message || e)); continue; }
   585→    let n = walker.currentNode;
   586→    while (n) {
   587→      nodes++;
   588→      if (schedNeedYield()) await schedSlice();

   590→        if (!seen.has(n.shadowRoot)) { seen.add(n.shadowRoot); roots.push(n.shadowRoot); stack.push(n.shadowRoot); }
   591→      }
   592→      if (nodes >= CFG.SR_NODE_CAP || Date.now() - start > CFG.SR_TIME_MS) { truncated = true; break; }
   593→      n = walker.nextNode();
   594→    }
   595→    if (truncated) break;
   596→  }
   597→  return { roots, truncated, nodes, ms: Date.now() - start, evidence: mkEv('treewalker', roots.length ? 0.9 : 0.4, ['nodes:' + nodes, truncated ? 'truncated' : 'complete']) };

   670→function candidateFeatures(els, platform, areaRoot) {
   671→  const count = els.length;
   672→  let known = 0, users = 0, assistants = 0, ui = 0, inArea = 0, nonEmpty = 0, lenSum = 0;
   673→  const sigs = new Map();
   674→  for (const el of els) {
   675→    const re = extractRoleRaw(el, platform);
   676→    if (re.role === 'user') { users++; known++; } else if (re.role === 'assistant') { assistants++; known++; }
   677→    const sig = el.tagName + '|' + (typeof el.className === 'string' ? el.className.split(' ')[0] : '');
   678→    sigs.set(sig, (sigs.get(sig) || 0) + 1);
   679→    if (isUiZone(el)) ui++;
   680→    if (areaRoot && (el === areaRoot || areaRoot.contains(el))) inArea++;
   681→    const len = (el.textContent || '').trim().length;
   682→    if (len > 0) nonEmpty++;
   683→    lenSum += len;
   684→  }
   685→  let modeFreq = 0; for (const v of sigs.values()) modeFreq = Math.max(modeFreq, v);
   686→  let nested = 0;
   687→  for (const el of els) { for (const other of els) { if (other !== el && other.contains(el)) { nested++; break; } } }
   688→  const avg = count ? lenSum / count : 0;
   689→  return {
   690→    count,
   691→    roleCoherence: count ? known / count : 0,
   692→    membership: areaRoot ? inArea / count : 0.5,
   693→    uniformity: count ? modeFreq / count : 0,
   694→    orderConsistency: count >= 3 ? ((users > 0 && assistants > 0) ? 1 : 0.2) : 0.4,
   695→    textQuality: count ? Math.min(1, (nonEmpty / count) * (avg >= 8 ? 1 : avg / 8)) : 0,
   696→    nestedShare: count ? nested / count : 0,
   697→    uiShare: count ? ui / count : 0,
   698→    cleanliness: count ? 1 - Math.min(1, ui / count) : 0,
   699→    users, assistants
   700→  };
   701→}
