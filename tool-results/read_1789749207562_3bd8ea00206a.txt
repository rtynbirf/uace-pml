   364→    const roots = [];

   368→    if (cached) return cached.slice();

   370→      if (!r || seen.has(r)) return;

   545→      if (node.nodeType === 3) return node.nodeValue || '';
   546→      if (node.nodeType !== 1) return '';
   547→      const tag = node.tagName.toLowerCase();
   548→      if (tag === 'script' || tag === 'style' || tag === 'svg' || tag === 'noscript') return '';

   551→      if (tag === 'pre') {

   554→        const text = (code ? code.textContent : node.textContent) || '';
   555→        return '\n\n```' + lang + '\n' + String(text).replace(/\r\n/g, '\n').replace(/\n+$/, '') + '\n```\n\n';
   556→      }
   557→      if (tag === 'code') return '`' + clean(node.textContent) + '`';

   559→      if (tag === 'a') { const href = node.getAttribute('href') || ''; const v = clean(children(node)); return href ? '[' + (v || href) + '](' + href + ')' : v; }
   560→      if (/^h[1-6]$/.test(tag)) return '\n\n' + '#'.repeat(Number(tag[1])) + ' ' + clean(children(node)) + '\n\n';
   561→      if (tag === 'strong' || tag === 'b') { const v = clean(children(node)); return v ? '**' + v + '**' : ''; }
   562→      if (tag === 'em' || tag === 'i') { const v = clean(children(node)); return v ? '*' + v + '*' : ''; }
   563→      if (tag === 'del' || tag === 's') { const v = clean(children(node)); return v ? '~~' + v + '~~' : ''; }
   564→      if (tag === 'blockquote') { const v = clean(children(node)); return '\n\n' + v.split('\n').map((l) => '> ' + l).join('\n') + '\n\n'; }
   565→      if (tag === 'li') { const ordered = node.parentElement && node.parentElement.tagName.toLowerCase() === 'ol'; return (ordered ? '1. ' : '- ') + clean(children(node)) + '\n'; }
   566→      if (tag === 'ul' || tag === 'ol') return '\n\n' + children(node) + '\n';
   567→      if (tag === 'table') return tableToMarkdown(node);
   568→      if (tag === 'p' || tag === 'div' || tag === 'section' || tag === 'article' || tag === 'figure') return '\n' + children(node) + '\n';

   582→    }

   623→  function scrape(options) {
   624→    options = options || {};
   625→    const found = findMessages();
   626→    if (!found) return { items: [], platform: null, source: null, images: [], attachments: [] };
   627→    const platform = found.platform;
   628→    const pre = [];
   629→    found.items.forEach((el) => {
   630→      const content = stripUiPrefix(domToMarkdown(el));
   631→      const text = stripUiPrefix(extractText(el));
   632→      if (!content && !text) return;
   633→      const plain = text || content;
   634→      if (plain.length < CFG.MIN_TEXT) return;
   635→      if (isCssOrJs(plain) || isSourceListing(plain)) return;
   636→      pre.push({ el: el, content: content, text: text, thinking: extractThinking(el) });
   637→    });
   638→    const roles = pre.map((m) => detectRole(m.el, platform));
   639→    const allUnknown = roles.length > 0 && roles.every((r) => r === 'unknown');
   640→    const allSame = roles.length > 1 && roles.every((r) => r === roles[0]);
   641→    const parity = allUnknown || (allSame && roles[0] !== 'user');
   642→    const allImages = [], allAttachments = [];
   643→    const out = pre.map((m, i) => {
   644→      let role = roles[i];
   645→      if (role === 'unknown' && parity) role = i % 2 === 0 ? 'user' : 'assistant';
   646→      else if (parity && role === roles[0]) role = i % 2 === 0 ? 'user' : 'assistant';
   647→      let content = m.content;
   648→      if (options.stripThinking && m.thinking.length) {
   649→        for (const block of m.thinking) content = content.replace(block, '');
   650→        content = clean(content);
   651→      }
   652→      const media = collectMedia(m.el, i);
   653→      media.images.forEach((x) => allImages.push(x));
   654→      media.attachments.forEach((x) => allAttachments.push(x));
   655→      return { role: role, content: content, text: m.text || clean(m.content), thinking: options.exportThinking ? m.thinking : [] };
   656→    }).filter((m) => (options.stripThinking ? m.content : m.text) && (options.stripThinking ? m.content.length : m.text.length) >= CFG.MIN_TEXT);
   657→    return { items: out.slice(0, CFG.MSG_MAX), source: found.source, platformName: (platform && platform.name) || null, images: allImages.slice(0, CFG.IMG_TOTAL), attachments: allAttachments.slice(0, CFG.IMG_TOTAL) };
   658→  }

   723→  async function waitStable(prevFp, prevHref) {
   724→    const start = Date.now();
   725→    let lastFp = '', lastChangeTime = Date.now();
   726→    let hrefChanged = !!(prevHref && location.href !== prevHref);
   727→    while (Date.now() - start < CFG.MAX_WAIT) {
   728→      if (CANCEL) return false;
   729→      await sleep(CFG.POLL);
   730→      if (!hrefChanged && prevHref && location.href !== prevHref) { hrefChanged = true; lastChangeTime = Date.now(); }
   731→      const cur = fingerprint();
   732→      if (cur && cur !== prevFp) {
   733→        if (cur !== lastFp) { lastFp = cur; lastChangeTime = Date.now(); }
   734→        else if (Date.now() - lastChangeTime >= CFG.STABLE_MS) return true;
   735→      } else if (hrefChanged && Date.now() - lastChangeTime >= CFG.STABLE_MS) {
   736→        return true;
   737→      }
   738→      if (!hrefChanged && !lastFp && Date.now() - start > 6000) return false;
   739→    }
   740→    const fp = fingerprint();
   741→    return (!!fp && fp !== prevFp) || hrefChanged;
   742→  }

   891→  const CRC_TABLE = (function () {
   892→    const t = new Uint32Array(256);
   893→    for (let n = 0; n < 256; n++) {
   894→      let c = n;
   895→      for (let k = 0; k < 8; k++) c = (c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1);
   896→      t[n] = c >>> 0;
   897→    }
   898→    return t;
   899→  })();
   900→  function crc32(u8) {
   901→    let c = 0xFFFFFFFF;
   902→    for (let i = 0; i < u8.length; i++) c = CRC_TABLE[(c ^ u8[i]) & 0xFF] ^ (c >>> 8);
   903→    return (c ^ 0xFFFFFFFF) >>> 0;
   904→  }
   905→  function dosDateTime(d) {
   906→    const time = ((d.getHours() << 11) | (d.getMinutes() << 5) | Math.floor(d.getSeconds() / 2)) & 0xFFFF;
   907→    const date = (((d.getFullYear() - 1980) & 0x7f) << 9) | ((d.getMonth() + 1) << 5) | d.getDate();
   908→    return { time: time, date: date };
   909→  }
   910→  // entries: [{name, data:string|Uint8Array, date?}] -> Blob (application/zip)
   911→  function zipStore(entries) {
   912→    const enc = new TextEncoder();
   913→    const localChunks = [], centralChunks = [];
   914→    let offset = 0;
   915→    const d = new Date();
   916→    const { time, date } = dosDateTime(d);
   917→    for (const e of entries) {
   918→      const name = enc.encode(e.name);
   919→      const data = (e.data instanceof Uint8Array) ? e.data : enc.encode(String(e.data));
   920→      const crc = crc32(data);
   921→      const lh = new DataView(new ArrayBuffer(30));
   922→      lh.setUint32(0, 0x04034b50, true);
   923→      lh.setUint16(4, 20, true);
   924→      lh.setUint16(6, 0x0800, true);
   925→      lh.setUint16(8, 0, true);
   926→      lh.setUint16(10, time, true);
   927→      lh.setUint16(12, date, true);
   928→      lh.setUint32(14, crc, true);
   929→      lh.setUint32(18, data.length, true);
   930→      lh.setUint32(22, data.length, true);
   931→      lh.setUint16(26, name.length, true);
   932→      lh.setUint16(28, 0, true);
   933→      localChunks.push(new Uint8Array(lh.buffer), name, data);
   934→      const ch = new DataView(new ArrayBuffer(46));
   935→      ch.setUint32(0, 0x02014b50, true);
   936→      ch.setUint16(4, 20, true);
   937→      ch.setUint16(6, 20, true);
   938→      ch.setUint16(8, 0x0800, true);
   939→      ch.setUint16(10, 0, true);
   940→      ch.setUint16(12, time, true);
   941→      ch.setUint16(14, date, true);
   942→      ch.setUint32(16, crc, true);
   943→      ch.setUint32(20, data.length, true);
   944→      ch.setUint32(24, data.length, true);
   945→      ch.setUint16(28, name.length, true);
   946→      ch.setUint16(30, 0, true);
   947→      ch.setUint16(32, 0, true);
   948→      ch.setUint16(34, 0, true);
   949→      ch.setUint16(36, 0, true);
   950→      ch.setUint32(38, 0, true);
   951→      ch.setUint32(42, offset, true);
   952→      centralChunks.push(new Uint8Array(ch.buffer), name);
   953→      offset += 30 + name.length + data.length;
   954→    }
   955→    let cdSize = 0;
   956→    for (const c of centralChunks) cdSize += c.length;
   957→    const eocd = new DataView(new ArrayBuffer(22));
   958→    eocd.setUint32(0, 0x06054b50, true);
   959→    eocd.setUint16(4, 0, true);
   960→    eocd.setUint16(6, 0, true);
   961→    eocd.setUint16(8, entries.length, true);
   962→    eocd.setUint16(10, entries.length, true);
   963→    eocd.setUint32(12, cdSize, true);
   964→    eocd.setUint32(16, offset, true);
   965→    eocd.setUint16(20, 0, true);
   966→    return new Blob(localChunks.concat(centralChunks, [new Uint8Array(eocd.buffer)]), { type: 'application/zip' });
   967→  }
