    99→    99→const srgbLin = c => (c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4));
   100→   100→const luminance = rgb => 0.2126 * srgbLin(rgb[0]) + 0.7152 * srgbLin(rgb[1]) + 0.0722 * srgbLin(rgb[2]);
   101→   101→const contrast = (a, b) => { const l1 = luminance(a), l2 = luminance(b); return (Math.max(l1, l2) + 0.05) / (Math.min(l1, l2) + 0.05); };
   102→   102→function rgbToOklch(rgb) {
   103→   103→  const [r, g, b] = rgb.map(srgbLin);
   104→   104→  const l = 0.4122214708 * r + 0.5363325363 * g + 0.0514459929 * b;
   105→   105→  const m = 0.2119034982 * r + 0.6806995451 * g + 0.1073969566 * b;
   106→   106→  const s = 0.0883024619 * r + 0.2817188376 * g + 0.6299787005 * b;
   107→   107→  const L = 0.2104542553 * Math.cbrt(l) + 0.7936177850 * Math.cbrt(m) - 0.0040720468 * Math.cbrt(s);
   108→   108→  const A = 1.9779984951 * Math.cbrt(l) - 2.4285922050 * Math.cbrt(m) + 0.4505937099 * Math.cbrt(s);
   109→   109→  const B = 0.0259040371 * Math.cbrt(l) + 0.7827717662 * Math.cbrt(m) - 0.8086757660 * Math.cbrt(s);
   110→   110→  const C = Math.sqrt(A * A + B * B);
   111→   111→  return { L, C, H: (Math.atan2(B, A) * 180 / Math.PI + 360) % 360 };
   112→   112→}
   113→   113→function oklchToRgb(L, C, H) {
   114→   114→  const h = H * Math.PI / 180, a = C * Math.cos(h), b = C * Math.sin(h);
   115→   115→  const l_ = L + 0.3963377774 * a + 0.2158037573 * b, m_ = L - 0.1055613458 * a - 0.0638541728 * b, s_ = L - 0.0894841775 * a - 1.2914855480 * b;
   116→   116→  const l = l_ ** 3, m = m_ ** 3, s = s_ ** 3;
   117→   117→  const f = x => clamp(x <= 0.0031308 ? 12.92 * x : 1.055 * Math.pow(x, 1 / 2.4) - 0.055, 0, 1);
   118→   118→  return [f(4.0767416621 * l - 3.3077115913 * m + 0.2309699292 * s),
   119→   119→          f(-1.2684380046 * l + 2.6097574011 * m - 0.3413193965 * s),
   120→   120→          f(-0.0041960863 * l - 0.7034186147 * m + 1.7076147010 * s)];
   121→   121→}
   122→   122→const rgbToHex = rgb => '#' + rgb.map(v => Math.round(v * 255).toString(16).padStart(2, '0')).join('');
   123→   123→const hexToRgb = hex => [1, 3, 5].map(i => parseInt(hex.slice(i, i + 2), 16) / 255);
   124→   124→function parseColor(str) {                      // "rgb(r, g, b)" / "#hex" → [r,g,b] 0..1, иначе null
   125→   125→  const m = /rgba?\(([^)]+)\)/.exec(str);
   126→   126→  if (m) { const p = m[1].split(',').map(parseFloat); return p.length >= 3 ? [p[0] / 255, p[1] / 255, p[2] / 255] : null; }
   127→   127→  const h = /^#[0-9a-f]{6}$/i.exec(str.trim());
   128→   128→  return h ? hexToRgb(h[0]) : null;
   129→   129→}
   130→   130→function fitAccent(bgRgb, sysRgb) {             // двухпроход OKLCH (мастер §6) → {hex, txHex, ratio}
   131→   131→  const sys = rgbToOklch(sysRgb);
   132→   132→  if (sys.C < 0.04 || sys.L < 0.05 || sys.L > 0.98) return null;   // сатурационный страж: серый/чёрный/белый
   133→   133→  let best = null;

   135→   135→    const rgb = oklchToRgb(L, Math.min(sys.C, 0.16), sys.H);

   137→   137→    if (r1 < 4.5) continue;

   143→   143→  }
