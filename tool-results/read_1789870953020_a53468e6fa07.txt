  2851→const PML_COLLECT_TIME_CAP_MS = 15;
  2852→function pmlCollectDeep() {
  2853→  const out = []; let total = 0;
  2854→  const tStart = performance.now();
  2855→  const stack = [document.body];
  2856→  const seenShadows = new Set();
  2857→  while (stack.length) {
  2858→    const cur = stack.pop();
  2859→    if (!cur || cur.nodeType !== 1) continue;
  2860→    if (SKIP_TAGS.has(cur.tagName) || pmlIsOwnNode(cur)) continue;
  2861→    total++;
  2862→    if (out.length < CAP.nodes) out.push(cur);
  2863→    try {
  2864→      const sr = cur.shadowRoot;
  2865→      if (sr && sr.firstElementChild && !seenShadows.has(sr) && (!sr.mode || sr.mode === 'open') && !pmlIsOwnNode(cur)) {
  2866→        seenShadows.add(sr);
  2867→        for (let c = sr.firstElementChild; c; c = c.nextElementSibling) stack.push(c);
  2868→      }
  2869→    } catch (e) {}
  2870→    for (let c = cur.firstElementChild; c; c = c.nextElementSibling) stack.push(c);
  2871→    if ((out.length >= CAP.nodes && total > CAP.nodes + 2048) || (performance.now() - tStart > PML_COLLECT_TIME_CAP_MS)) break;
  2872→  }
  2873→  out.total = total;

  2891→function pmlScoreGroup(g) {
