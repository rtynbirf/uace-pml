   165→  while (n && n.nodeType === 1 && n.tagName !== 'BODY' && n.tagName !== 'HTML') {
   166→    let i = 1, s = n;
   167→    while ((s = s.previousElementSibling)) if (s.tagName === n.tagName) i++;
   168→    segs.unshift(n.tagName.toLowerCase() + (i > 1 ? ':nth(' + i + ')' : ''));
   169→    n = n.parentElement;
   170→  }
   171→  return ['body'].concat(segs);
   172→};
   173→const pathOf = el => {                         // структурный путь ≤120 симв. (r1 §0): nth(i) только при i>1
   174→  const segs = pathSegs(el).slice(1);
   175→  let p = 'body' + (segs.length ? '>' + segs.join('>') : '');
   176→  if (p.length > 120) p = '~' + p.slice(-119); // усечение слева, хвост глубже важнее
   177→  return p;
   178→};
   179→const prefixPath = (el, d) => pathSegs(el).slice(0, d).join('>');
   180→const sigOf = el => {                          // r2 §5: pre-order теги ВЕРХНИМ регистром через '>', глубина ≤3
   181→  const out = [];
   182→  (function walk(n, d) {
   183→    if (!n || n.nodeType !== 1 || d > 3) return;
   184→    out.push(n.tagName.toUpperCase());
   185→    for (let c = n.firstElementChild; c; c = c.nextElementSibling) walk(c, d + 1);
   186→  })(el, 1);
   187→  return out.join('>');
   188→};
   189→const sigHash = el => hex8(FNV1a32(sigHashIn(sigOf(el))));
   190→const subtreeDepth = (el, cap) => {            // фактическая глубина поддерева (cap 6)
   191→  let max = 1;
   192→  (function walk(n, cur) {
   193→    if (cur > cap) return;
   194→    for (let c = n.firstElementChild; c; c = c.nextElementSibling) { max = Math.max(max, cur + 1); walk(c, cur + 1); }
   195→  })(el, 1);
   196→  return max;
   197→};
   198→const median = arr => {                        // медиана целых; null если пусто
   199→  const a = arr.filter(Number.isFinite).sort((x, y) => x - y);
   200→  if (!a.length) return null;
   201→  const m = a.length >> 1;
   202→  return a.length % 2 ? a[m] : Math.round((a[m - 1] + a[m]) / 2);
   203→};

   205→function pickPrimary(all, vh) {                // I-2: каскад primary (I.3.2) + контейнеры — столбец UPOLY
   206→  const cands = [];
   207→  for (const el of all) {
   208→    if (el.scrollHeight <= el.clientHeight + 4) continue;
   209→    let ov = 'visible';
   210→    try { ov = getComputedStyle(el).overflowY; } catch (e) {}
   211→    if (ov !== 'auto' && ov !== 'scroll' && ov !== 'overlay') continue;
   212→    let main = false;
   213→    try { main = !!el.closest('main,[role="main"]'); } catch (e) {}
   214→    cands.push({ el, path: pathOf(el), clientHeight: el.clientHeight, scrollHeight: el.scrollHeight, visibleShare: el.clientHeight / vh, insideMain: main });
   215→  }
   216→  cands.sort((a, b) => b.visibleShare - a.visibleShare || b.scrollHeight - a.scrollHeight || b.clientHeight - a.clientHeight || (b.insideMain - a.insideMain) || (a.path < b.path ? -1 : 1));
   217→  return { primary: cands[0] || null, containers: cands.slice(0, CAP.containers).map(c => ({ path: c.path, scrollHeight: c.scrollHeight, clientHeight: c.clientHeight })), truncated: cands.length > CAP.containers };
   218→}
   219→function findGroups(all, primaryPath, deadline) {   // I-1: модальная группа на родителя — столбец UPOLY (+ inPrimary r2 3-a)
   220→  const out = [];
   221→  for (const parent of all) {
   222→    if (performance.now() > deadline) { out.budgetHit = true; break; }
   223→    if (parent.childElementCount < 3) continue;
   224→    const kids = [...parent.children];
   225→    const bySig = new Map();
   226→    for (const k of kids) { const h = sigHash(k); if (!bySig.has(h)) bySig.set(h, []); bySig.get(h).push(k); }
   227→    let best = null;
   228→    for (const [h, arr] of bySig) if (!best || arr.length > best.arr.length) best = { h, arr };
   229→    if (!best || best.arr.length < 3) continue;
   230→    const u = best.arr.length / kids.length;
   231→    if (u < 0.6) continue;
   232→    let pr;
   233→    try { pr = parent.getBoundingClientRect(); } catch (e) { continue; }
   234→    let sumH = 0, ok = true;
   235→    for (const k of best.arr) {
   236→      const r = k.getBoundingClientRect();
   237→      if (r.height < 32) { ok = false; break; }   // I.2.3: каждый ребёнок ≥32px
   238→      sumH += r.height;
   239→    }
   240→    if (!ok || sumH < 0.4 * pr.height) continue;  // отсечение toolbar/иконочных рядов
   241→    out.push({ sigHash: best.h, count: best.arr.length, depth: subtreeDepth(best.arr[0], 6), uniformity: hx(u), samplePath: pathOf(best.arr[0]), parentPath: pathOf(parent), inPrimary: !!primaryPath && (pathOf(best.arr[0]) === primaryPath || pathOf(best.arr[0]).startsWith(primaryPath + '>')), el: best.arr[0] });
   242→  }
   243→  out.sort((a, b) => b.count * b.uniformity - a.count * a.uniformity || (a.parentPath < b.parentPath ? -1 : 1)); // I.2.5
   244→  return out;
   245→}
   246→function findMounts(all, vh, deadline) {            // I-4a: кандидаты монтирования — столбец UPOLY + свои узлы вне
   247→  const out = [];
   248→  for (const el of all) {
   249→    if (performance.now() > deadline) { out.budgetHit = true; break; }
   250→    if (host && (el === host || host.contains(el))) continue;   // свои узлы исключены
   251→    if (el.childElementCount < 1) continue;
   252→    let cs; try { cs = getComputedStyle(el); } catch (e) { continue; }
   253→    if (cs.display === 'none' || cs.visibility === 'hidden') continue;
   254→    if (cs.position !== 'fixed' && cs.position !== 'sticky') continue;
   255→    const r = el.getBoundingClientRect();
   256→    if (!(r.top < 0.15 * vh && r.height >= 32 && r.height <= 96 && r.width >= 120)) continue;
   257→    out.push({ path: pathOf(el), mode: cs.position, h: Math.round(r.height), top: r.top, x: r.x });
   258→  }
   259→  out.sort((a, b) => a.top - b.top || a.x - b.x);
   260→  return out;
   261→}
   262→function findLandmarks(all) {                   // C3 (позиция B на r2): структурные теги + ландмарк-роли, без role=button
   263→  const out = [];

   265→  const roles = ['banner', 'main', 'navigation', 'complementary', 'contentinfo', 'search', 'region', 'form'];
   266→  for (const el of all) {
   267→    const role = el.getAttribute('role');
   268→    if (!tags[el.tagName] && roles.indexOf(role) < 0) continue;
   269→    out.push({ tag: el.tagName.toLowerCase(), role: role || null, path: pathOf(el) });
   270→  }
   271→  return out;

   273→function scanSync() {                           // один обход от body (свой хост вне body — самоисключение), бюджет ≤150 мс
   274→  const t0 = performance.now();
   275→  const bodyEl = document.body;
   276→  if (!bodyEl) return { all: [], vh: innerHeight || 800, blockCount: 0, maxDepth: 0, sc: { primary: null, containers: [], truncated: false }, groups: [], mounts: [], landmarks: [], truncatedDom: false, elapsed: 0 };
   277→  const every = bodyEl.querySelectorAll('*');

   279→  const vh = innerHeight || 800;
   280→  let blockCount = 0, maxDepth = 0;
   281→  (function w(n, d) { if (!n || n.nodeType !== 1) return; blockCount++; maxDepth = Math.max(maxDepth, d); for (let c = n.firstElementChild; c; c = c.nextElementSibling) w(c, d + 1); })(bodyEl, 1);

   290→function scoreGroup(g) {                        // r2 §2: чистая формула «кандидат → вес-факты»
   291→  const u = Math.round(100 * Math.max(0, Math.min(1, (g.uniformity - 0.6) / 0.4)));
   292→  const c = Math.round(100 * Math.min(1, g.count / 40));
   293→  const p = g.inPrimary ? 100 : 0;
   294→  const s = Math.round(100 * Math.max(0, Math.min(1, 1 - g.ratio)));
   295→  const l = g.hasLandmark ? 100 : 0;
   296→  return { u, c, p, s, l, confidence: Math.round(0.35 * u + 0.25 * c + 0.20 * p + 0.10 * s + 0.10 * l) };
   297→}
   298→function resolveRecon(scored) {                 // r2 §3: гейты G1–G4 → found/no-data
   299→  const ok = scored.filter(x => x.inPrimary && x.count >= 6 && x.depth >= 2).sort((a, b) => b.confidence - a.confidence || (a.samplePath < b.samplePath ? -1 : 1));
   300→  if (!ok.length || ok[0].confidence < 75) return { status: 'no-data' };
   301→  const second = ok[1];
   302→  if (second && ok[0].confidence - second.confidence < 15 && second.confidence >= 50) return { status: 'no-data' };
   303→  const t = ok[0];
   304→  return { status: 'found', candidate: { path: t.samplePath, sigHash: t.sigHash, inPrimary: true, confidence: t.confidence, basis: { u: t.u, c: t.c, p: t.p, s: t.s, l: t.l } } };
   305→}
   306→function buildStruct() {                        // конверт r1 (+inPrimary, recon по r2) — форма столбца UPOLY

   318→    dom: { maxDepth: s.maxDepth, blockCount: s.blockCount, landmarks: s.landmarks.slice(0, CAP.landmarks), groups: raw.map(g => ({ sigHash: g.sigHash, count: g.count, depth: g.depth, uniformity: g.uniformity, samplePath: g.samplePath, inPrimary: g.inPrimary })), truncated: s.truncatedDom || s.groups.length > CAP.groups || s.landmarks.length > CAP.landmarks },

   321→    anchors: { mountPoints: s.mounts.slice(0, CAP.mountPoints).map(m => ({ path: m.path, mode: m.mode, h: m.h })), persistentPaths: obsResult ? obsResult.persistent : [], stability: obsResult ? obsResult.stability : 0, truncated: s.mounts.length > CAP.mountPoints || (obsResult ? obsResult.persistentTruncated : false) }

   349→const downsample = (st, budget) => { // D2Snap (arXiv 08.2025): coverage + selection под бюджет; детерминированно
   350→  const B = Math.max(60, budget == null ? 400 : budget | 0), items = [];
   351→  const push = (prio, line) => items.push({ prio, line, cost: Math.ceil(line.length / 4) }); // «токен» ≈ 4 символа
   352→  push(0, 'page ' + ((st.meta && st.meta.urlHash) || '?'));
   353→  ((st.dom && st.dom.landmarks) || []).forEach((l, i) => push(1, (l.role || l.tag) + ' [ref=l' + (i + 1) + '] ' + l.path));
   354→  ((st.anchors && st.anchors.mountPoints) || []).forEach((m, i) => push(2, 'mount ' + m.mode + ' h' + m.h + ' ' + m.path));
   355→  ((st.dom && st.dom.groups) || []).forEach((g, i) => push(3, 'group ' + g.sigHash + ' x' + g.count + ' d' + g.depth + (g.inPrimary ? ' *' : '') + ' ' + g.samplePath));
   356→  items.sort((a, b) => a.prio - b.prio); // стабильная сортировка: внутри приоритета — канонный порядок struct
   357→  const kept = []; let used = 0, dropped = 0;
   358→  for (const it of items) if (used + it.cost <= B) { kept.push(it.line); used += it.cost; } else dropped++;
   359→  if (dropped) kept.push('… +' + dropped + ' узлов за бюджетом ' + B);
   360→  return { view: kept.join('\n'), used, budget: B, dropped };

   362→const compareStructs = (a, b) => { // Broder/minhash-линейка (1997): точный Jaccard по sigHash групп (наборы ≤12)
   363→  const ga = new Set(((((a || {}).dom || {}).groups) || []).map(g => g.sigHash));
   364→  const gb = new Set(((((b || {}).dom || {}).groups) || []).map(g => g.sigHash));
   365→  let inter = 0; for (const h of ga) if (gb.has(h)) inter++;

   367→  return { jaccard: union ? hx(inter / union) : 1, sim: simBetween(a, b), inter, groupsA: ga.size, groupsB: gb.size };

   369→/* ---- I-3 + I-4b: пассивное окно 60 с; Esc — стоп; снапшот 0 + каждые 10 с (столбец UPOLY) ---- */
   370→function startObservation() {
   371→  if (obsSession) return;
   372→  if (!document.body) { log('err', 'нет document.body — наблюдение отменено'); status('наблюдение: нет body'); return; }
   373→  metrics.obsRuns++;
   374→  const s = { adds: 0, removes: 0, recycles: 0, sigBudget: 500, aborted: false, byPath: new Map(), addIntervals: new Map(), lastAdd: new Map(), removedSigs: new Map(), snapshots: [], navEvents: 0, pausedMs: 0, hiddenAt: 0 };
   375→  s.endsAt = Date.now() + OBS_WINDOW_MS;
   376→  obsSession = s;
   377→  setBadge('run');
   378→  status('Наблюдение: осталось 60 с');
   379→  log('info', 'окно наблюдения 60 с запущено (Esc — стоп)');
   380→  s.observer = new MutationObserver(muts => {
   381→    for (const m of muts) {
   382→      const p = m.target && m.target.nodeType === 1 ? prefixPath(m.target, 4) : 'body';
   383→      let rec = s.byPath.get(p);
   384→      if (!rec) { rec = { adds: 0, removes: 0, net: 0 }; s.byPath.set(p, rec); }
   385→      for (const n of m.addedNodes) if (n.nodeType === 1) {
   386→        s.adds++; rec.adds++; rec.net++;
   387→        const t = performance.now(), last = s.lastAdd.get(p);
   388→        if (last != null) { const a = s.addIntervals.get(p) || []; a.push(Math.round(t - last)); s.addIntervals.set(p, a); }
   389→        s.lastAdd.set(p, t);
   390→        if (s.sigBudget > 0) {
   391→          s.sigBudget--;
   392→          const h = sigHash(n), set = s.removedSigs.get(p);
   393→          if (set && set.has(h)) { s.recycles++; set.delete(h); }   // I.4.4: remove→add с равным sigHash
   394→        }
   395→      }
   396→      for (const n of m.removedNodes) if (n.nodeType === 1) {
   397→        s.removes++; rec.removes--; rec.net--;
   398→        if (s.sigBudget > 0) { const set = s.removedSigs.get(p) || s.removedSigs.set(p, new Set()).get(p); if (set.size < 50) set.add(sigHash(n)); }
   399→      }
   400→    }
   401→  });
   402→  s.observer.observe(document.body, { childList: true, subtree: true, characterData: false, attributes: false });
   403→  const snap = () => { const set = new Set(); (function w(n, d) { if (!n || n.nodeType !== 1 || d > 4 || set.size >= 300) return; if (!host || n !== host) { set.add(pathOf(n)); for (let c = n.firstElementChild; c; c = c.nextElementSibling) w(c, d + 1); } })(document.body, 1); s.snapshots.push(set); };
   404→  s.snap = snap; snap();                         // I.5.2: снапшот 0 + каждые 10 с, глубина ≤4, cap 300
   405→  s.snapTimer = setInterval(() => { if (!document.hidden && s.snapshots.length < 30) snap(); }, 10000);
   406→  s.navHandler = () => { s.navEvents++; snap(); };
   407→  s.op = history.pushState; s.orp = history.replaceState;
   408→  history.pushState = function () { s.navHandler(); return s.op.apply(this, arguments); };
   409→  history.replaceState = function () { s.navHandler(); return s.orp.apply(this, arguments); };
   410→  addEventListener('popstate', s.navHandler);
   411→  s.visHandler = () => { if (document.hidden) s.hiddenAt = Date.now(); else if (s.hiddenAt) { s.pausedMs += Date.now() - s.hiddenAt; s.hiddenAt = 0; } };
   412→  document.addEventListener('visibilitychange', s.visHandler);
   413→  s.tick = setInterval(() => {
   414→    const left = Math.max(0, Math.ceil((s.endsAt + s.pausedMs - Date.now()) / 1000));
   415→    status('Наблюдение: осталось ' + left + ' с');
   416→    if (left <= 0) finishObservation(false);
   417→  }, 1000);
   418→}
   419→function finishObservation(aborted, silent) {
   420→  const s = obsSession;
   421→  if (!s) return;
   422→  obsSession = null;
   423→  clearInterval(s.tick); clearInterval(s.snapTimer);
   424→  s.observer.disconnect();
   425→  document.removeEventListener('visibilitychange', s.visHandler);
   426→  removeEventListener('popstate', s.navHandler);
   427→  history.pushState = s.op; history.replaceState = s.orp;
   428→  const top = [...s.byPath.entries()].sort((a, b) => b[1].adds - a[1].adds)[0];
   429→  const med = top ? median(s.addIntervals.get(top[0]) || []) : null;
   430→  const stream = top ? (top[1].adds >= 3 && top[1].net > 0 && med !== null && med >= 50 && med <= 5000) : false; // I.4.3
   431→  obsResult = {
   432→    mut: { window: 60, adds: s.adds, removes: s.removes, recycles: s.recycles, medianIntervalMs: med, streamCandidate: stream },
   433→    persistent: [], stability: 0, persistentTruncated: false
   434→  };
   435→  if (s.snapshots.length) {                      // I.5.2: присутствуют в ≥90% снапшотов
   436→    const total = s.snapshots.length, cnt = new Map();
   437→    for (const set of s.snapshots) for (const p of set) cnt.set(p, (cnt.get(p) || 0) + 1);
   438→    const per = [...cnt.entries()].filter(x => x[1] / total >= 0.9).sort((a, b) => b[1] - a[1] || (a[0] < b[0] ? -1 : 1));
   439→    obsResult.persistent = per.slice(0, CAP.persistentPaths).map(x => x[0]);
