  1321→function sigOfEl(el) { /* pre-order сигнатура глубиной ≤3, UPPERCASE (канон §5) */
  1322→  const out = [];
  1323→  (function walk(n, d) { if (!n || n.nodeType !== 1 || d > 3) return; out.push(n.tagName.toUpperCase()); for (let c = n.firstElementChild; c; c = c.nextElementSibling) walk(c, d + 1); })(el, 1);
  1324→  return out.join('>');
  1325→}
  1326→const subtreeDepth = (el, cap) => { let max = 1; (function walk(n, cur) { if (cur > cap) return; for (let c = n.firstElementChild; c; c = c.nextElementSibling) { max = Math.max(max, cur + 1); walk(c, cur + 1); } })(el, 1); return max; };

  1334→function pmlPickPrimary(all, vh) {
  1335→  const cands = [];
  1336→  for (const el of all) {
  1337→    if (el.scrollHeight <= el.clientHeight + 4) continue;
  1338→    let ov = 'visible'; try { ov = getComputedStyle(el).overflowY; } catch (e) { journal.debug('pml', 'overflow-fail', String(e && e.message || e)); }
  1339→    if (ov !== 'auto' && ov !== 'scroll' && ov !== 'overlay') continue;

  1342→  }
  1343→  cands.sort((a, b) => b.visibleShare - a.visibleShare || b.scrollHeight - a.scrollHeight || b.clientHeight - a.clientHeight || (b.insideMain - a.insideMain) || (a.path < b.path ? -1 : 1));
  1344→  return { primary: cands[0] ? { path: cands[0].path, ratio: round2(cands[0].clientHeight / Math.max(1, cands[0].scrollHeight)), clientHeight: cands[0].clientHeight, scrollHeight: cands[0].scrollHeight } : null, containers: cands.slice(0, PML_CAP.containers).map(c => ({ path: c.path, scrollHeight: c.scrollHeight, clientHeight: c.clientHeight })), truncated: cands.length > PML_CAP.containers };
  1345→}
  1346→function pmlFindGroups(all, primaryPath, deadline) {
  1347→  const out = [];
  1348→  for (const parent of all) {
  1349→    if (performance.now() > deadline) { out.budgetHit = true; break; }
  1350→    if (parent.childElementCount < 3) continue;
  1351→    const bySig = new Map();
  1352→    for (const k of parent.children) { const h = fnv1aHex(sigHashIn(sigOfEl(k))); if (!bySig.has(h)) bySig.set(h, []); bySig.get(h).push(k); }
  1353→    let best = null;
  1354→    for (const [h, arr] of bySig) if (!best || arr.length > best.arr.length) best = { h, arr };
  1355→    if (!best || best.arr.length < 3) continue;
  1356→    const u = best.arr.length / parent.childElementCount;
  1357→    if (u < 0.6) continue;
  1358→    let pr; try { pr = parent.getBoundingClientRect(); } catch (e) { continue; }
  1359→    let sumH = 0, ok = true;
  1360→    for (const k of best.arr) { const r = k.getBoundingClientRect(); if (r.height < 32) { ok = false; break; } sumH += r.height; }
  1361→    if (!ok || sumH < 0.4 * pr.height) continue;
  1362→    const samplePath = pathOf(best.arr[0]);
  1363→    out.push({ sigHash: best.h, count: best.arr.length, depth: subtreeDepth(best.arr[0], 6), uniformity: round2(u), samplePath, parentPath: pathOf(parent), inPrimary: !!primaryPath && (samplePath === primaryPath || samplePath.startsWith(primaryPath + '>')) });
  1364→  }
  1365→  out.sort((a, b) => b.count * b.uniformity - a.count * a.uniformity || (a.parentPath < b.parentPath ? -1 : 1));
  1366→  return out;
  1367→}
  1368→function pmlFindMounts(all, vh, deadline) {
  1369→  const out = [];
  1370→  for (const el of all) {
  1371→    if (performance.now() > deadline) { out.budgetHit = true; break; }
  1372→    if (el.closest && el.closest('#' + UI_HOST_ID)) continue;
  1373→    if (el.childElementCount < 1) continue;
  1374→    let cs; try { cs = getComputedStyle(el); } catch (e) { continue; }
  1375→    if (cs.display === 'none' || cs.visibility === 'hidden') continue;
  1376→    if (cs.position !== 'fixed' && cs.position !== 'sticky') continue;
  1377→    const r = el.getBoundingClientRect();
  1378→    if (!(r.top < 0.15 * vh && r.height >= 32 && r.height <= 96 && r.width >= 120)) continue;
  1379→    out.push({ path: pathOf(el), mode: cs.position, h: Math.round(r.height), top: Math.round(r.top), x: Math.round(r.x) });
  1380→  }
  1381→  out.sort((a, b) => a.top - b.top || a.x - b.x);
  1382→  return out;
  1383→}
  1384→function pmlFindLandmarks(all) {
  1385→  const out = [];
  1386→  const tags = { MAIN: 1, NAV: 1, HEADER: 1, FOOTER: 1, ASIDE: 1, SECTION: 1 };
  1387→  for (const el of all) {

  1389→    if (!tags[el.tagName] && !role) continue;
  1390→    out.push({ tag: el.tagName.toLowerCase(), role: role || null, path: pathOf(el) });
  1391→    if (out.length > PML_CAP.landmarks * 2) break;
  1392→  }
  1393→  return out.slice(0, PML_CAP.landmarks);
  1394→}
  1395→/* Скоринг r2 §2: u/c/p/s/l = 0.35/0.25/0.20/0.10/0.10; резерв r2 §3 (порог 75) */
  1396→function scoreGroup(g, s) {
  1397→  const u = Math.round(100 * clamp((g.uniformity - 0.6) / 0.4, 0, 1));
  1398→  const c = Math.round(100 * clamp((g.count - 3) / 17, 0, 1));
  1399→  const p = g.inPrimary ? 100 : 0;
  1400→  const sz = Math.round(100 * clamp((s ? s.primary && s.primary.ratio || 0 : 0), 0, 1));
  1401→  const l = Math.round(100 * clamp(g.depth / 6, 0, 1));
  1402→  return { u, c, p, s: sz, l, confidence: Math.round(0.35 * u + 0.25 * c + 0.20 * p + 0.10 * sz + 0.10 * l) };
  1403→}

  1405→async function pmlScan() {
  1406→  const t0 = performance.now();
  1407→  const all = await pmlCollectElements();
  1408→  const vh = innerHeight || 800;
  1409→  const primary = pmlPickPrimary(all, vh);
  1410→  const groups = pmlFindGroups(all, primary.primary ? primary.primary.path : null, t0 + PML_CAP.scanMs);
  1411→  const mounts = pmlFindMounts(all, vh, t0 + PML_CAP.scanMs);
  1412→  const landmarks = pmlFindLandmarks(all);
  1413→  let blockCount = 0, maxDepth = 1;
  1414→  const w = document.createTreeWalker(document.body, NodeFilter.SHOW_ELEMENT);
  1415→  let n = w.currentNode, depth = 1;
  1416→  while (n && blockCount < 4000) { blockCount++; n = w.nextNode(); }
  1417→  let deep = document.body, d = 1;
  1418→  while (deep && deep.firstElementChild && d < 30) { deep = deep.firstElementChild; d++; }
  1419→  maxDepth = d;
  1420→  const dom = { maxDepth, blockCount, landmarks, groups, truncated: !!(groups.budgetHit || mounts.budgetHit) || blockCount >= 4000 };
  1421→  const candidate = groups.length ? (() => { const g = groups[0]; const sc = scoreGroup(g, primary); return { sigHash: g.sigHash, count: g.count, uniformity: g.uniformity, confidence: sc.confidence, passed: sc.confidence >= 75 }; })() : null;
  1422→  lastStruct = {

  1425→    dom, anchors: { mountPoints: mounts },
  1426→    recon: candidate ? { status: candidate.passed ? 'found' : 'weak', candidate } : { status: 'none', candidate: null },
  1427→    capture: { ms: Math.round(performance.now() - t0), nodes: all.length }
  1428→  };
