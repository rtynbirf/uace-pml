   260→const NavGuard = {
   261→  wrapped: false,
   262→  origPush: null,
   263→  origReplace: null,
   264→  subs: new Set(),
   265→  ensure() {
   266→    if (this.wrapped) return true;

   268→      const hp = history.pushState, hr = history.replaceState;
   269→      if (typeof hp !== 'function' || typeof hr !== 'function') return false;
   270→      const guard = this;
   271→      history.pushState = function () { guard._notify(); return hp.apply(this, arguments); };
   272→      history.replaceState = function () { guard._notify(); return hr.apply(this, arguments); };
   273→      guard.origPush = hp; guard.origReplace = hr; guard.wrapped = true;
   274→      return true;
   275→    } catch (e) { return false; }
   276→  },
   277→  _notify() { for (const fn of Array.from(this.subs)) { try { fn(); } catch (e) {} } },
   278→  subscribe(fn) {
   279→    if (typeof fn !== 'function' || this.subs.has(fn)) return false;
   280→    this.ensure();
   281→    this.subs.add(fn);
   282→    return true;
   283→  },
   284→  unsubscribe(fn) { return this.subs.delete(fn); }

   513→function collectShadowRoots(root) {
   514→  const roots = []; const seen = new Set();
   515→  const nowMs = Date.now();
   516→  const cached = (!root && _srCache && (nowMs - _srCacheAt) < 1500) ? _srCache : null;
   517→  if (cached) return cached.slice();
   518→  /* v28.2: полностью итеративный обход (стек) — рекурсия устранена (риск stack overflow);
   519→     закрытые (closed) shadowRoot недоступны и/или mode!=='open' — пропускаем осознанно */
   520→  const stack = [root || document];
   521→  seen.add(root || document);
   522→  let visited = 0;
   523→  while (stack.length) {
   524→    const cur = stack.pop();
   525→    if (!cur || (cur.nodeType !== 9 && cur.nodeType !== 1 && cur.nodeType !== 11)) continue;
   526→    let all; try { all = cur.querySelectorAll('*'); } catch (e) { continue; }
   527→    for (let i = 0; i < all.length; i++) {
   528→      if (++visited > SR_NODE_CAP) { stack.length = 0; break; }
   529→      const el = all[i];
   530→      let sr = null;
   531→      try { sr = el.shadowRoot; } catch (e) {}
   532→      if (!sr || seen.has(sr)) continue;
   533→      if (sr.mode && sr.mode !== 'open') { seen.add(sr); continue; }
   534→      seen.add(sr); roots.push(sr); stack.push(sr);
   535→    }
   536→  }
   537→  if (!root) { _srCache = roots; _srCacheAt = nowMs; }
   538→  return roots;
   539→}

  1288→async function poolMap(items, limit, worker, statusTick) {

  1293→    while (cursor < arr.length) {
  1294→      const idx = cursor++;
  1295→      try { out[idx] = { ok: true, value: await worker(arr[idx], idx) }; }
  1296→      catch (e) { out[idx] = { ok: false, error: e };
  1297→      } finally { done++; if (statusTick) try { statusTick(done, arr.length); } catch (e) {} }
  1298→    }
  1299→  });
