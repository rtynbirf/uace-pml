   593→const SR_NODE_CAP = 12000; // v28.2: защита от сверхтяжёлых DOM
   594→function collectShadowRoots(root) {
   595→  const roots = []; const seen = new Set();
   596→  const nowMs = Date.now();
   597→  const cached = (!root && _srCache && (nowMs - _srCacheAt) < 1500) ? _srCache : null;
   598→  if (cached) return cached.slice();
   599→  /* v28.2: полностью итеративный обход (стек) — рекурсия устранена (риск stack overflow);
   600→     закрытые (closed) shadowRoot недоступны и/или mode!=='open' — пропускаем осознанно */
   601→  const stack = [root || document];
   602→  seen.add(root || document);
   603→  let visited = 0;
   604→  while (stack.length) {
   605→    const cur = stack.pop();
   606→    if (!cur || (cur.nodeType !== 9 && cur.nodeType !== 1 && cur.nodeType !== 11)) continue;
   607→    let all; try { all = cur.querySelectorAll('*'); } catch (e) { continue; }
   608→    for (let i = 0; i < all.length; i++) {
   609→      if (++visited > SR_NODE_CAP) { stack.length = 0; break; }
   610→      const el = all[i];
   611→      let sr = null;
   612→      try { sr = el.shadowRoot; } catch (e) {}
   613→      if (!sr || seen.has(sr)) continue;
   614→      if (sr.mode && sr.mode !== 'open') { seen.add(sr); continue; }
   615→      seen.add(sr); roots.push(sr); stack.push(sr);
   616→    }
   617→  }
   618→  if (!root) { _srCache = roots; _srCacheAt = nowMs; }
   619→  return roots;
   620→}
