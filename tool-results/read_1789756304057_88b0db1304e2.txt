   912→  async function waitStable(prevFp, prevHref) {
   913→    const start = Date.now();
   914→    let lastFp = '', lastChangeTime = Date.now();
   915→    let hrefChanged = !!(prevHref && location.href !== prevHref);
   916→    while (Date.now() - start < CFG.MAX_WAIT) {
   917→      if (CANCEL) return false;
   918→      await sleep(CFG.POLL);
   919→      if (!hrefChanged && prevHref && location.href !== prevHref) { hrefChanged = true; lastChangeTime = Date.now(); }
   920→      const cur = fingerprint();
   921→      if (cur && cur !== prevFp) {
   922→        if (cur !== lastFp) { lastFp = cur; lastChangeTime = Date.now(); }
   923→        else if (Date.now() - lastChangeTime >= CFG.STABLE_MS) return true;
   924→      } else if (hrefChanged && Date.now() - lastChangeTime >= CFG.STABLE_MS) {
   925→        return true;
   926→      }
   927→      if (!hrefChanged && !lastFp && Date.now() - start > 6000) return false;
   928→    }
   929→    const fp = fingerprint();
   930→    return (!!fp && fp !== prevFp) || hrefChanged;
   931→  }
