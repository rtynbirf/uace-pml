    20→  function hfTheme() {
    21→    const t = safeStorageGet('uaice_theme') || 'auto';
    22→    if (t === 'light' || t === 'dark') return t;
    23→    // auto: вечер после 20:00 и утро до 07:00 — тёплый режим (синий свет/мелатонин, Harvard)
    24→    const evening = safeStorageGet('uaice_evening') !== '0';
    25→    const h = new Date().getHours();
    26→    if (evening && (h >= 20 || h < 7)) return 'warm';
    27→    try { if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) return 'dark'; } catch (e) {}
    28→    return 'light';
    29→  }
    30→  function hfMotionOn() {
    31→    if (safeStorageGet('uaice_motion') === '0') return false;
    32→    try { if (window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches) return false; } catch (e) {}
    33→    return true;
    34→  }
    35→  // WCAG-контраст живьём — паспорт панели считает сам, не ссылается на чужие обещания
    36→  function hfLum(hex) {
    37→    const c = hex.replace('#', '');
    38→    const v = [0, 2, 4].map((i) => {
    39→      let x = parseInt(c.slice(i, i + 2), 16) / 255;
    40→      return x <= 0.03928 ? x / 12.92 : Math.pow((x + 0.055) / 1.055, 2.4);
    41→    });
    42→    return 0.2126 * v[0] + 0.7152 * v[1] + 0.0722 * v[2];
    43→  }
    44→  function hfContrast(fg, bg) {
    45→    const l1 = hfLum(fg), l2 = hfLum(bg);
    46→    return ((Math.max(l1, l2) + 0.05) / (Math.min(l1, l2) + 0.05));
    47→  }
